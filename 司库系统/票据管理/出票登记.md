public void updateBillStatusByBillIds(String billIds, String billStatus, String operationAble, String stockFlag, String outStockTime){
    String[] billIdArr = billIds.split(Constant.DELIMITER_ANGLE);
    // 查询票据信息
    List<LdgBillAccountDTO> dbList = ConvertUtils.sourceToTarget(ldgBillAccountDao.findByIds(billIdArr),
            LdgBillAccountDTO.class);
    if (CollUtil.isEmpty(dbList)) {
        return;
    }
    for (LdgBillAccountDTO dto : dbList) {
        dto.setBillStatus(billStatus);
        dto.setOperationAble(operationAble);
        dto.setStockFlag(stockFlag);
        dto.setOutStockTime(DateUtil.parse(outStockTime,DatePattern.NORM_DATETIME_FORMAT));
        dto.setLastUpdateDate(new Date());
        // 新加逻辑：新票据平台，提交托收申请审批通过转已托收时，电票票据状态=已结清，流动标识=已结束
        if (StrUtil.equals(dto.getTicketSystem(), LedgerEnum.TICKET_SYSTEM.NEW.getCode()) && StrUtil.equals(
                billStatus, EnumType.BillStatus.BILL_STATUS_006.getValue())) {
            dto.setNewBillStatus(LedgerEnum.NEW_BILL_STATUS.NEW_BILL_STATUS_006.getCode());
            dto.setCirculateFlag(LedgerEnum.CIRCULATE_FLAG.CIRCULATE_FLAG_09.getCode());
        }
    }
    List<LdgBillAccountEntity> entityList = ConvertUtils.sourceToTarget(dbList, LdgBillAccountEntity.class);
    String sql = "UPDATE ZFS_LDG.LDG_BILL_ACCOUNT SET BILL_STATUS =? , OPERATION_ABLE =?, STOCK_FLAG =?, OUT_STOCK_TIME =?, NEW_BILL_STATUS =?, CIRCULATE_FLAG =?, LAST_UPDATE_DATE =? WHERE ID =? ";
    List<String> fieldNameList = new ArrayList<>(4);
    fieldNameList.add(BILL_STATUS);
    fieldNameList.add(OPERATION_ABLE);
    fieldNameList.add(STOCK_FLAG);
    fieldNameList.add(OUT_STOCK_TIME);
    fieldNameList.add(NEW_BILL_STATUS);
    fieldNameList.add(CIRCULATE_FLAG);
    fieldNameList.add(LAST_UPDATE_DATE);
    fieldNameList.add(ID);
    this.jdbcBatchUpdate(sql, entityList, fieldNameList);
}
