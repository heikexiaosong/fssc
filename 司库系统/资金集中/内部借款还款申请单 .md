@Transactional(rollbackFor = Exception.class)
public void doExecuteApproveEndRule(BoeFormDto boeFormDto, List<Object> extendVariableList,
                                    Map<String, Object> extendParams, String serviceName) {
    logger.info("内部借款还款申请单 结束动作开始");
    // 内部借款还款申请单审批完成后，
    // 若是否支付字段为“是”，则需要自动在付款受理生成一条或者两条付款受理的草稿态单据（根据内部借款还款申请单上还款本金和还款利息分别各生成一条付款受理单，金额为0则无需生成）
    // fmsFoeLedgerHandle/savePayBoeInfo
    String boePaymentId = zfsBoeCommonDaoFmsClient.savePayBoeInfo(boeFormDto);
    
    logger.info("内部借款还款申请单审批完成后生成付款受理草稿成功，付款受理ID为：" + boePaymentId);
    // 写入内部借款经办台账
    // fmsFoeLedgerHandle/externalFoeApproveEndRuleAndPay
    zfsBoeCommonDaoFmsClient.externalFoeApproveEndRuleAndPay(boeFormDto, boePaymentId);
}
