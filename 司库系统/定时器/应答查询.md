- 最新需求：【【同仁堂司库】银企分开部署改造--银企模块】https://www.tapd.cn/tapd_fe/67410840/story/detail/1167410840001324180



UPDATE zfs_fms.fms_provide_header t
INNER JOIN zfs_sys.sys_bank_united_code u ON u.ID = t.BANK_ID
SET
    t.FINANCIAL_INSTITUTION_CODE = COALESCE(NULLIF(TRIM(t.FINANCIAL_INSTITUTION_CODE), ''), u.UNITED_CODE),
    t.FINANCIAL_INSTITUTION = COALESCE(NULLIF(TRIM(t.FINANCIAL_INSTITUTION), ''), u.BANK_HEAD_OFFICE)
WHERE t.BANK_MARK = '0' -- 银行标识 = 银行
  and (NULLIF(TRIM(t.FINANCIAL_INSTITUTION_CODE), '') IS NULL OR NULLIF(TRIM(t.FINANCIAL_INSTITUTION), '') IS NULL)
  AND (NULLIF(TRIM(u.UNITED_CODE), '') IS NOT NULL OR NULLIF(TRIM(u.BANK_HEAD_OFFICE), '') IS NOT NULL);
