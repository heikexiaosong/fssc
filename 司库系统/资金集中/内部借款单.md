  from  zfs_ldg.ldg_bill_account ldgbillacc0_
where  ldgbillacc0_.use_le_id = '82135b6777a14bce874ab1b7b2ae129a'
	and ldgbillacc0_.postal_order_code like '%26110414%'
	and (ldgbillacc0_.finance_type is null or ldgbillacc0_.finance_type = '')
	and ldgbillacc0_.pool_type = '1'

	and ldgbillacc0_.pool_id = '1747b7a3e54af198a4c73137322e6605'
