for  dto in result:
    FmsBalanceDayDTO fmsDto= ObjUtil.newInstance(FmsBalanceDayDTO.class);
    fmsDto.setBankNumber( dto.getActacn());
    
    String curcde = StrUtil.isBlank(dto.getCurcde()) ? CNY : dto.getCurcde();
    fmsDto.setCurrencyCode(curcde);
   
    String currType = dto.getCurrType();
    if (StrUtil.isNotBlank(currType)) {
        # 币种映射(当银行返回的币种编码和当前系统币种存在映射关系)
        List<FmsCurrencycodeRelationEntity> entities1 = relationBankCodeMaps.get(currType);
        if (CollUtil.isNotEmpty(entities1)) {
            FmsCurrencycodeRelationEntity entity = entities1.get(0);
            String currencyCode = entity.getCurrencyCode();
            fmsDto.setCurrencyCode(currencyCode);
        }
    }
    
    String key= dto.getActacn() + Constant.DELIMITER_SUBTRACT_ESCAPE + dto.getCurcde();
    if(distinct_map.containsKey(key)) {
        // 根据账号+币种 去重
        continue;
    }
    fmsDto.setBookBalance(dto.getBokbal());
    fmsDto.setAvailableBalance(dto.getAvabal());
    if(fmsDto.getBookBalance() == null && dto.getRegularAmount() != null) {
        // 账面余额为空，定期余额不为空赋值给账面余额
        fmsDto.setBookBalance(dto.getRegularAmount());
    }
    fmsDto.setRegisterDate(new Date());
    
    
    distinct_map.put(key, key);
    fmsBalanceDayDTOList.add(fmsDto);

resultMap.put("list",fmsBalanceDayDTOList);
