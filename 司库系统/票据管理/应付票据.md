# 应付票据

### 获取票据详情
-【FMS】 https://127.0.0.1/fmspayablebill/3e46a92645ad8f9eb5773137322e0889

```
LdgBillAccountDTO dto = coreLdgClient.findByIdForFms(id);
map.put("ldgBillAccount",baseDataTranslateUtils.baseDataTranslate(dto, LdgBillAccountDTO.class));

// 需要查询出票据对应的附件
fmsAttachmentEntityList = fmsAttachmentService.findByBusIdAndType(id, FmsConstants.busType.BILL.getCode());
map.put("fmsAttachment",fmsAttachmentDTOs);

// 获取变更记录
billChangeList = fmsBillChangeHisService.findByBillId(id);
map.put("billChangeList",fmsBillChangeHisDTOs);

// 核算主体占用金额
leUseAmountRecords = fmsReceivableBillService.findLeUseAmountRecords(dto);
map.put("leUseAmountRecordS",leUseAmountRecords);

// 设置保证金信息
fmsBillDepositInfoService.fillBillDepositInfo(boeFormDto);
map.put("fmsBillDepositInfoDTOS",boeFormDto.getFmsBillDepositInfoDTOS());

// 设置费用信息
fmsBillFeeInfoService.fillBillFeeInfo(boeFormDto);
map.put("fmsBillFeeInfoDTOS",boeFormDto.getFmsBillFeeInfoDTOS());


// 查询&设置 自定义字段
FmsLedgerTemplateDTO ledgerTemplateDTO = ObjUtil.newInstance(FmsLedgerTemplateDTO.class);
//参数：模板编码和业务域
ledgerTemplateDTO.setBusinessDomainCode(LedgerConfigEnums.BUSINESS_DOMAIN_CODE.BILL_MANAGEMENT.getCode());
ledgerTemplateDTO.setLedgerCode(LedgerConfigEnums.LEDGER_CODE.PAYABLE_BILL.getCode());
extendMap = fmsLedgerDataExtendService.findByLedgerIdAndFieldCode(dto.getId(), ledgerTemplateDTO);
map.put("extendMap", extendMap);

```
