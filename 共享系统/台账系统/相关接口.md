from ldg_contract t

where t.ENABLED_FLAG = '0'
  and e.ENABLED_FLAG = '0'
  and t.CONTRACT_STATUS = '2'
  and t.contract_Scope = 'Y'
  and t.VENDOR_ID like '%8fef6f9f13c86e710da53139322e6661%'
  
union all

select t.ID, t.GROUP_ID, t.DRAFT_USER_ID, t.POST_ID, t.DEPT_ID
     , t.CONTRACT_TYPE, t.CONTRACT_CODE, t.CONTRACT_NAME, t.FRAMEWORK_CONTRACT, t.CONTRACT_AMOUNT

     , e.UNSETTLED_AMOUNT, e.CONFIRMED_INVOICE_AMOUNT, t.BEGIN_INVOICED_AMOUNT, e.AMOUNT_TRANSIT, e.UNCONFIRMED_AMOUNT

     , t.EXTEND_N10
from ldg_contract t

where t.ENABLED_FLAG = '0'
  and e.ENABLED_FLAG = '0'
  and t.CONTRACT_STATUS = '2'
  and t.contract_Scope = 'N'

    select au.CONTRACT_ID
    from ldg_contract_auths au
    where au.AUTH_EMP_ID in ('2112356661f7aa260abf3139322e06a6', '2112356661f7aa269e683139322e06aa', 'a3fd0af6c9370ee098613139322e0010', '8fef6f9f2678ef7a015f3139322e6df4')
      and au.AUTH_TYPE = '001'
      and au.ENABLED_FLAG = '0'
      and au.VALIDITY_FLAG = '0'
)
  and t.VENDOR_ID like '%8fef6f9f13c86e710da53139322e6661%'
