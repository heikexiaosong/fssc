# 付方层级信息[付方层级信息.png]



1. 【fmsInnerAccounts】找到内部户信息
```sql
select * 
  from fms_inner_account ia 
       left join fms_Pool_Org po on ia.org_id = po.id
       left join fms_pool_org_language fpo_lang on po.ID = fpo_lang.ORG_ID AND fpo_lang.LANGUAGE = :language
 where ia.ID in :nbId
```

2. 【fmsPoolOrgEntity】通过内部户的核算主体找资金组织关联表在通过关联表找到资金组织
```sql
select * 
  from FMS_POOL_REORG fpr
 where fpr.ORG_TYPE = '2' -- 机构类型: 1-资金池 2-内部户
   and fpr.LE_ID = :leId

select *
  from FMS_POOL_ORG
 where id = FMS_POOL_REORG::ORG_ID
```

3. 【accountPoolOrgEntity】查询银行账户资金组织
```sql
select * 
  from FMS_ACCOUNT_MAPPING fam
 where fam.INNER_ACCOUNT_ID = :nbId

select *
  from FMS_ACCOUNT fa
 where fa.id = FMS_ACCOUNT_MAPPING[0]::OUT_ACCOUNT_ID

-- 根据账户的法人实体ID和集团ID获取基金组织实体 FMS_ACCOUNT::LE_ID
-- 通过第二部逻辑查找
```

#### type == "0"
付款当前资金组织: currentFundsOrgf = fmsInnerAccounts.getOrgName()
付款办理资金组织: handlFundOrgf = fmsPoolOrgEntity.getId()
银行账户资金组织: accountHandleFundOrgf = accountPoolOrgEntity.getId()

#### type == "1"
收款当前资金组织: currentFundsOrgs = fmsInnerAccounts.getOrgName()
收款办理资金组织: handlFundOrgs = fmsPoolOrgEntity.getId()
银行账户资金组织: accountHandleFundOrgs = accountPoolOrgEntity.getId()

### 【collect】层级信息
```java
// accountMap: FMS_INNER_ACCOUNT<ID, FMS_INNER_ACCOUNT>
// convert: 资金组织的FNO字段上保存的所有上级资金组织列表
FmsPaymentAndReceiptHierarchyInformationDTO entity = null;
// 构建层级数据
for (int j = 0; j < convert.size(); j++) {
    entity = ObjUtil.newInstance(FmsPaymentAndReceiptHierarchyInformationDTO.class);
    entity.setSort(j);
    if (CharSequenceUtil.isNotBlank(currentNbId)) {
        FmsInnerAccountEntity fmsInnerAccountEntity = accountMap.get(currentNbId);
        if (ObjectUtil.isNotEmpty(fmsInnerAccountEntity)) {
            // 设置内部户账号层级数据
            entity.setAccountHierarchyfundOrgId(fmsInnerAccountEntity.getId()); // 资金组织内部账户ID
            entity.setAccountHierarchyfundOrgBn(fmsInnerAccountEntity.getBankNumber()); // 资金组织内部账户账号
            entity.setAccountHierarchyfundOrgName(fmsInnerAccountEntity.getAccountName()); // 资金组织内部账户名称
            entity.setOrgId(fmsInnerAccountEntity.getOrgId()); // 资金组织id
            // 更新下一个要查询的ID
            currentNbId = fmsInnerAccountEntity.getpAccountId() ? null
        } else {
            currentNbId = null;
        }
    }
    listNbh.add(entity);
}
```

### 【list】组装资质组织数据
- cn.ztessc.service.boe.FmsPaymentApplyHeaderService#assembleQualificationOrganizationData
```java
// filterFmsPoolOrgDTOs: 资金组织以及所有的上级资金组织列表
// orgIdMap: collect<orgId, new FmsPaymentAndReceiptHierarchyInformationDTO<FmsInnerAccountEntity>>
int level = 0;
for (FmsPoolOrgDTO orgDTO : filterFmsPoolOrgDTOs) {
    level = level + 1;

    entity = ObjUtil.newInstance(FmsPaymentAndReceiptHierarchyInformationDTO.class);
    // 设置基本信息
    entity.setType(fmsPaymentAndReceiptHierarchyInformationDTO.getType());
    entity.setBoeId(fmsPaymentAndReceiptHierarchyInformationDTO.getBoeId());
    cn.ztessc.common.utils.EntityUtils.setWho(entity);

    // 将索引转换为中文数字(从1开始)
    String numberToChinese = Convert.numberToChinese(level, false);

    // 确定层级类型前缀
    String typePrefix = params::type == "0" ? "付款" : "收款";    
    // 设置层级信息
    entity.setFundOrgHierarchy(typePrefix + numberToChinese + "级资金组织");
    entity.setFundOrgId(orgDTO.getId());
    entity.setFundOrgCode(orgDTO.getCode());
    entity.setFundOrgName(orgDTO.getOrgName());

    // 设置账户层级数据(如果存在)
    //  orgIdMap: collect<orgId, new FmsPaymentAndReceiptHierarchyInformationDTO<FmsInnerAccountEntity>>
    FmsPaymentAndReceiptHierarchyInformationDTO targetDto = orgIdMap.getOrDefault(orgDTO.getId(), new);
    entity.setAccountHierarchyfundOrg(typePrefix + numberToChinese + "级内部账户");
    entity.setAccountHierarchyfundOrgId(targetDto.getAccountHierarchyfundOrgId());
    entity.setAccountHierarchyfundOrgBn(targetDto.getAccountHierarchyfundOrgBn());
    entity.setAccountHierarchyfundOrgName(targetDto.getAccountHierarchyfundOrgName());

    entity.setLevel(level);
    list.add(entity);
}
```
