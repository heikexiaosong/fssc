  from fms_bill_sign_info fmsbillsig0_
 where fmsbillsig0_.validity_flag = '0'
	 and fmsbillsig0_.sign_type in ('1', '2', '3', '4', '5', '6', '7')
	 and fmsbillsig0_.postal_order_code like '%630410004262620260703000672311%'
	 and fmsbillsig0_.show_flag = 'Y'
	 and fmsbillsig0_.enabled_flag = '0'
	 and fmsbillsig0_.group_id = 'af4ce58b58c721b9dcde3137322e0004'
  order by fmsbillsig0_.create_date desc, fmsbillsig0_.id desc
