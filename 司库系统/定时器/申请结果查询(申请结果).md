-- 查询指令状态为 STATUS_3("3", "发送成功") 的指令
select * 
  from FMS_ELE_BILL_INSTRUCTIONS t
 where t.INSTRUCTION_STATUS = '3'
   and t.VALIDITY_FLAG ='0' 
   and t.ENABLED_FLAG ='0'

select * 
  from FMS_ACCOUNT t

   and (t.SALES_MARK is null or t.SALES_MARK = '0') -- 销户标志(正常、销户).

```python
for (FmsEleBillInstructionsEntity entity : value):
  fmsAccount = fmsAccountMap.get(entity.getFmsAccountId());

  bankHeadId = fmsAccount.getBankHeadId();
  postalOrderCode = entity.getPostalOrderCode();
  if postalOrderCode == '' && bankHeadId not in ["SK_LOGO_TY", "PAB_HEADER_BANK_CODE", "bocHeaderBankCode", "icbcHeaderBankCode", "cmbHeaderBankCode"]:
    continue
  

  // 请求参数组装
  paramDto = ObjUtil.newInstance(QueryMoneyOrderApplyResultParamDto.class);
 
  routeGroupCode = groupCode;
  if (companyCodeAppendEnabled) {
      routeGroupCode = fmsAccountService.groupCodeAppend(groupCode, companyCodeMap, userId, fmsAccount.getLeId());
  }
  paramDto.setGroupCode(routeGroupCode);
  
  paramDto.setBankCode(fmsAccount.getBankHeadId());  
  paramDto.setEnterpriseAccount(fmsAccount.getBankNumber());
  paramDto.setDraftNb(entity.getPostalOrderCode());
  paramDto.setInstructionNo(entity.getInstructionNo());
  paramDto.setApplyDate(entity.getApplyDate());

  if (StrUtil.isNotBlank(entity.getCltkey())) {
      // 中行存储的obssid，业务字段
      paramDto.setCltKey(entity.getCltkey());
  }

  if bankHeadId == 'cmbHeaderBankCode':
    // 招商银行
    paramDto.setDueDtBegin(DateUtil.offsetMonth(DateUtil.date(), -1));
    paramDto.setDueDtEnd(new Date());
    paramDto.setRequestBatchNumber(entity.getMsgId());

  if bankHeadId == 'bocHeaderBankCode':
    // 中国银行
    if instructionType == "9":
      // 出票申请 清空票据号 
      paramDto.setDraftNb("");

  if bankHeadId == 'citicHeaderBankCode':
    // 中信银行
    // 调用新方法处理电票指令
    processBillInstructionPoolStatus(value, fmsAccountMap, companyCodeAppendEnabled, companyCodeMap);
    continue;

  if bankHeadId == 'SK_LOGO_TY' && fmsAccount.getBankHeadId() == "105":
    // 天元司库 建设银行
    paramDto.setApplyDate(entity.getDateOfApplication());
    paramDto.setCltKey(entity.getBankSerialNo());


  if instructionType == "TYPE_4":
    // 背书申请
    paramDto.setQueryTp(FmsEnumType.QUERY_TYPE_WH.QUERY_TYPE_WH_4.getValue());
  if instructionType == "TYPE_7":
    // 贴现申请
    paramDto.setQueryTp(FmsEnumType.QUERY_TYPE_WH.QUERY_TYPE_WH_5.getValue());

  // TODO ...

  retDto = BankRouteHandle.queryMoneyOrderApplyResultObject(bankHeadId, paramDto, groupCode);

```



### 相关信息
- 需求： 【【电票直连】票据池招商银行接口】https://www.tapd.cn/tapd_fe/67410840/story/detail/1167410840001274405
【账户管理】： 
  POOL_CONNECTION 开通/未开通; 
  POOL_CONNECTION_CHANNEL 票据池直连通道. 01：银行通道；   02： 天元司库；  03： 票交所;

FMS参数：SAME_GROUP_COMPANY_SEPARATE_USB_KEY
   判断是否启用“同集团公司是否分开不同U盾”功能（即是否需拼接 COMPANY_CODE）. 
   值为 "Y"（严格字符串相等）→ 启用拼接
   其他情况（null、空白、非Y值、异常）→ 不启用（安全降级）
   相关需求： 【【同仁堂司库】银企分开部署改造--银企模块】 https://www.tapd.cn/tapd_fe/67410840/story/detail/1167410840001324180
