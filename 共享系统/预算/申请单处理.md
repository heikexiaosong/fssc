调用预算接口查询数据： bgtfeeapplyledgerexecution/feeApplyLedgerPage
```sql
-- 预算中查询sql
select SOURCE_BOE_ID, SOURCE_BOE_NUM, EMPLOYEE_ID, BOE_AMOUNT, OPERATION_TYPE_ID, AMOUNT, FREEZE_AMOUNT
from (
      select T.SOURCE_BOE_ID,
             T.SOURCE_BOE_NUM,
             T.EMPLOYEE_ID,
             T.BOE_AMOUNT,
             T.OPERATION_TYPE_ID,
             sum(T.AMOUNT) AMOUNT,
             ABS(sum(case when BOE_ASSOCIATE_STATUS = '2000' then AMOUNT else 0 end)) as USED_AMOUNT,
             ABS(sum(case when BOE_ASSOCIATE_STATUS = '1000' then AMOUNT else 0 end)) as USE_IN_TRANSIT_AMOUNT,
             ABS(sum(case when BOE_ASSOCIATE_STATUS = '3000' then AMOUNT else 0 end)) as CLOSE_AMOUNT,
             ABS(sum(case when BOE_ASSOCIATE_STATUS is null then AMOUNT else 0 end))  as FREEZE_AMOUNT
        from BGT_FEE_APPLY_LEDGER_EXECUTION T
       where T.CLOSE_STATUS = '1'
         and T.VALIDITY_FLAG = '0'
         and T.ENABLED_FLAG = '0'
         and (TRANSFER_STATUS <> '0' or TRANSFER_STATUS is null)
         and T.EMPLOYEE_ID = '21123566c0e1be2400563139322e6971'
         and T.OPERATION_TYPE_ID = '8fef6f9f7cb73bc718f23139322e18fd'
         and T.BOE_TYPE_ID in ('e54691360b8ffbd33fcbb1b0aeb5005d', '8fef6f9f007a7cc83f7e3139322e576d')
         and T.SOURCE_BOE_NUM like '%ZCJT-BX2609110075%'
         and T.ASSOCIATION_STATUS <> '1'
      group by T.SOURCE_BOE_NUM, T.EMPLOYEE_ID, T.BOE_AMOUNT, T.SOURCE_BOE_ID, T.OPERATION_TYPE_ID
    ) tmp
where tmp.FREEZE_AMOUNT - USED_AMOUNT - USE_IN_TRANSIT_AMOUNT + CLOSE_AMOUNT > 0 
limit 10
```
