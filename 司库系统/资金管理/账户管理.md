```python
# 账户销户申请单 经办登记 信息查询
# cn.ztessc.service.account.FmsAccountHandleService#hanldLines
def hanldLines(boeFormDto: BoeFormDto,  fmsAccountDto: FmsAccountDTO,  operationTypeCode: String)：
    Map<String, Object> result = DtoMapConverter.convertDtoToMapAuto(fmsAccountDto);

    List<Map<String, Object>> accountUsbList = new ArrayList<>();
    List<Map<String, Object>> accountSignatureSealList = new ArrayList<>();
    for line in boeFormDto.getZfsBoeLines():
        if line.lineType in ('accountUsbList', 'u'):
            accountUsbList.append(line.extendMap);        
        elif line.lineType in ('accountSignatureSealList', 'sealManagementArea'):
            accountSignatureSealList.append(line.extendMap);        
        elif line.lineType in ('ob'):
            result.update(line.extendMap);

        if line.lineType in ('fmsBase') and operationTypeCode != 'ZHSQ':
            result["id"] = line.bankAccountId
        elif line.lineType in ('fmsBase') and operationTypeCode == 'ZHSQ':
            result.setdefault(line.extendMap)  # 只有当 key 不在 result中时，才会把 k:v 放进去
            result["accountCountryRegion"] = extendMap["accountCountryRegionNew"] ? result["accountCountryRegion"]
            result["accountCountryRegionName"] = extendMap["accountCountryRegionNewName"] ? result["accountCountryRegionNewName"]
        
    result["accountUsbList"] = accountUsbList;
    result["accountSignatureSealList"] = accountSignatureSealList;   
    result["boeTemplateId"] = boeFormDto.boeHeader.boeTemplateId;  

    return result；
```
