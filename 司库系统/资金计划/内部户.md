### 付款申请单审批完成生成【内部户明细帐】
- cn.ztessc.service.boe.FmsPaymentApplyHeaderService#insertInnerAccountInfoForPaymentApply

```java
FmsInteriorDetailEntity detailEntity = ObjUtil.newInstance(FmsInteriorDetailEntity.class);
detailEntity.setId(UuidUtils.getUuid());
// 本方账号=收款/付款内部户账号
detailEntity.setInnerAccountId(innerAccountId);
// 交易类型
detailEntity.setCollectType(collectType);
// 币种=付款申请单-交易币种
detailEntity.setCurrencyCode(paymentApplyHeaderDTO.getPaymentCurrency());
// 业务日期=付款申请单生成的支付指令支付成功日期 -- 内部转账结算方式，业务日期取单据完成时间
detailEntity.setBusinessDate(paymentDate);
// 对方账号=付款/收款内部户账号，对方账户名称=付款/收款内部户账号名称
detailEntity.setOppositeBankNumber(oppositeBankNumber);
detailEntity.setOppositeBankName(oppositeBankName);
// 1支出=付款方展示贷方金额，付款申请单-金额  2收入=收款方展示借方金额，付款申请单-金额
if (StrUtil.equals("1", type)) {
    // 1支出=付款方展示贷方金额，付款申请单-金额
    detailEntity.setCreditAmount(paymentApplyHeaderDTO.getApplyAmount());
    detailEntity.setDebitAmount(BigDecimal.ZERO);
} else if (StrUtil.equals("2", type)) {
    // 2收入=收款方展示借方金额，付款申请单-金额
    detailEntity.setCreditAmount(BigDecimal.ZERO);
    detailEntity.setDebitAmount(paymentApplyHeaderDTO.getApplyAmount());
}
// 摘要=付款申请单-摘要字段
detailEntity.setRemark(paymentApplyHeaderDTO.getBoeAbstract());
// 内部业务单编号=取付款申请单单号
detailEntity.setBusinessBoeNo(paymentApplyHeaderDTO.getBoeNo());
detailEntity.setIsSync(FmsEnumType.INNER_IS_SYNC.UNSYNC.getValue());
detailEntity.setGroupId(paymentApplyHeaderDTO.getGroupId());
detailEntity.setBusId(paymentApplyHeaderDTO.getId());
detailEntity.setSourceSystem(FmsEnumType.SOURCE_SYSTEM.FMS.getCode());
detailEntity.setBusinessType(paymentApplyHeaderDTO.getBusinessType());
detailEntity.setBoeNo(paymentApplyHeaderDTO.getSourceBoeNo());
detailEntity.setBoeTypeCode(paymentApplyHeaderDTO.getBoeTypeCode());
EntityUtils.setWho(detailEntity);
fmsInteriorDetailEntityList.add(detailEntity);     
```
