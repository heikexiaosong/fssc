select t0.ID, t0.ORG_PERMISSION_TYPE, t0.SITES_TYPE, t0.MODE_ALL_FALG, t0.MENU_ALL_FALG
        , t0.LE_ALL_FALG, t0.ACCOUNT_ALL_FALG, t0.ORG_ALL_FALG, t0.BELONG_LE_ALL_FALG
  from fms_account_auth_config t0, fms_auth_config_emp t1, fms_auth_config_type t2
 where t0.id = t1.auth_config_id
   and t1.auth_config_id = t2.auth_config_id

   and t2.auth_type = '14'
   and t0.enabled_flag = '0'
   and t0.validity_flag = '0'

select r.*, i.loan_code, i.loan_status, i.lender_le_id, i.loaner_le_id
        , i.loan_way, ir.interest_way, i.loan_begin_date, i.loan_end_date, ir.billing_month
        , ir.billing_day, i.lender_inner_id, i.loaner_inner_id, ir.disbursement_detail_code, ir.last_interest_date
        , ir.last_withhold_date, i.credit_amount_sum, i.repay_amount_sum, ir.first_interest_date_type 
        , ir.first_interest_custom_date, ir.effective_date
  from FMS_INNER_LOAN_INTEREST_RECORD r
       inner join FMS_INNER_LOAN_LEDGER i on r.LOAN_ID = i.id
       inner join FMS_INNER_LOAN_LEDGER_RECORD ir on ir.ID = r.LOAN_RECORD_ID and ir.enabled_flag = '0'
 where i.enabled_flag = '0'
   and i.group_id = 'af4ce58b58c721b9dcde3137322e0004'
   and i.loan_code like '%HBZ0105107-260710-01%'
   and i.LOAN_WAY not in ('9', '10')
   and i.LOAN_STATUS in ('2', '3', '4')
   and r.END_DATE > ir.MAX_INTEREST_DATE
   and r.BEGIN_DATE < TIMESTAMP '2026-07-21 00:00:00.000'
   
   and ir.IS_INTEREST = '1'
   and ir.INTEREST_WAY in ('1', '04', '05', '06', '07')
   and ir.MAX_INTEREST_DATE < TIMESTAMP '2026-07-21 00:00:00.000'
   and (i.LOAN_END_DATE <= TIMESTAMP '2026-07-21 00:00:00.000' or (ir.BILLING_MONTH like '%7%' and ir.BILLING_DAY = '21'))
   and r.rate != 0
order by i.loan_code asc, ir.disbursement_detail_code asc, r.interest_type asc, r.begin_date asc
