### 资金上收
- https://127.0.0.1/fmsCollectDown/page
```json
{
  "bankNumber": "511608011018010060636",
  "orgPermissionType": "1",
  "fmsCollectType": "1"
}
```

```sql
select
	G.*
from
	FMS_COLLECT_DOWN G
  left join FMS_ACCOUNT A on G.ACCOUNT_ID = A.ID
	left join FMS_ACCOUNT B on G.COLLECT_ACCOUNT_ID = B.ID
where g.fms_collect_type = '1'
	and a.bank_number like '%511608011018010060636%'
	and g.group_id = 'af4ce58b58c721b9dcde3137322e0004'
	and g.validity_flag = '0'
	and g.enabled_flag = '0'
limit 20

-- 余额数据
select * 
from fms_balance_day fbd 
where ACCOUNT_ID  = '007ce205041ea7379187b1b7b2ae0952'

-- 在途数据
select sum(sd.AMOUNT),  85630423.40 - sum(sd.AMOUNT) as amount from fms_in_transit_amt sd
where BANK_ID  = '007ce205041ea7379187b1b7b2ae0952' and ENABLED_FLAG  = '0';
```
