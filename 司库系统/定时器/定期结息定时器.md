# 定期结息定时器

### batchDepositSettlement
```sql
-- list：查询状态为已放款/逾期/展期；是否结息=是；计息方式包含定期结息及新增方式；是否自动结息=是；结息日为昨日且昨日未手动结息（可结息总利息大于0）的内部借款
select	r.*,	i.loan_code,	i.loan_status,	i.lender_le_id,	i.loaner_le_id ,	i.loan_way,	ir.interest_way,	i.loan_begin_date,	i.loan_end_date,	ir.billing_month ,	ir.billing_day,	i.lender_inner_id,	i.loaner_inner_id,	ir.disbursement_detail_code,	ir.last_interest_date,	ir.last_withhold_date,	i.credit_amount_sum,	i.repay_amount_sum,	i.unrepay_amount
  from FMS_INNER_LOAN_INTEREST_RECORD r -- 内部借款利息记录
       inner join FMS_INNER_LOAN_LEDGER i on r.LOAN_ID = i.id -- 内部借款台账
       inner join FMS_INNER_LOAN_LEDGER_RECORD ir on ir.ID = r.LOAN_RECORD_ID -- 内部借款台账操作记录
 where i.enabled_flag = '0'
	 and r.rate != 0 -- 利率
	 and ir.IS_INTEREST = '1'  -- [内部借款台账操作记录]是否计息 0-不计息，1-计息
	 and ir.INTEREST_WAY in ('1', '04', '05', '06', '07') -- [内部借款台账操作记录]计息方式
	 and i.LOAN_STATUS in ('2', '3', '4')  -- [内部借款台账]业务状态 1-已登记、2-已放款、3-逾期、4-展期、5-结清、0-已废弃
	 and r.END_DATE > ir.MAX_INTEREST_DATE -- [内部借款利息记录]中止日 > [内部借款台账操作记录]结息日最大日期(用于存款结息查询)
	 and ir.MAX_INTEREST_DATE < TIMESTAMP '2026-09-05 00:00:00.000' -- [内部借款台账操作记录]结息日最大日期(用于存款结息查询)
	 and r.BEGIN_DATE < TIMESTAMP '2026-09-05 00:00:00.000' -- [内部借款利息记录]起始日
	 and (i.LOAN_END_DATE <= TIMESTAMP '2026-09-05 00:00:00.000' 
		    or (ir.BILLING_MONTH like '%9%' and ir.BILLING_DAY = '5')
		    or i.CREDIT_AMOUNT_SUM = i.REPAY_AMOUNT_SUM)  -- [内部借款利息记录]终止日期小于当日， [内部借款台账操作记录]结息包含9月5日, [内部借款台账]放款金额总和 = 还款金额总和
order by i.loan_code asc, ir.disbursement_detail_code asc, r.interest_type asc, r.begin_date asc
limit 100
```
