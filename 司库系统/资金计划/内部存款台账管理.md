--  查询台账信息
select * from FMS_INNER_DEPOSIT_LEDGER  where id = 'c22eae6b242f832ebb35b1b7b2ae15ea'

--  校验台账核算主体下的 活期账户 
select * 
from fms_inner_account fav 
where 1 = 1 
and fav.group_id = 'af4ce58b58c721b9dcde3137322e0004' 
and fav.le_id = '712eea2f6f8d723b207ab1b7b2ae16b1' 
and fav.validity_flag = '0' 
and fav.enabled_flag = '0' 
and fav.account_state = '0' 
and fav.account_type in ('01')   // 活期账户
and exists ( 
        select fac.currency_code 
                from fms_inner_currency fac 
        where fav.id = fac.account_id 
                        and fac.currency_code = 'CNY' 
        ) 
        and fav.id in ('f9f191860a6f926538f5b1b7b2ae1e3f') 
order by fav.create_date desc, fav.last_update_date desc 
limit 10
