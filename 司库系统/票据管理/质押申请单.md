- 3.  拆成子票，同时修改母票状态为“已拆分”
```sql
-- 查询票据业务表（查询金额信息busAmount）: billBusinessEntities
select * from FMS_BILL_BUSINESS where BOE_ID = :boeId and BILL_ID in :entityList::billId
```

billBusinessEntityMap: billBusinessEntities<billId, FmsBillBusinessEntity>

billList: coreLdgClient.findListByIds(entityList::billId);

for billDto in billList:
  ticketAmount = billDto.getTicketAmount(); # 票据开票金额

  FmsBillBusinessEntity bizEntity = billBusinessEntityMap.get(billDto.getId())；
  
  busAmount = bizEntity.getBusAmount(); # 质押金额
  if busAmount < ticketAmount:
    # 如果质押金额＜出票金额,（票据只质押了部分金额， 需要拆票）
    
    // 1. 已质押子票
    subBill = ConvertUtils.sourceToTarget(billDto, LdgBillAccountDTO.class);
    subBill.setId(UuidUtils.getUuid());
    subBill.setPoolId(null);
    subBill.setPoolType(null);

    subBill.setTicketAmount(bizEntity.getBusAmount());
    subBill.setSubticketStart(bizEntity.getPledgeAmountStart());
    subBill.setSubticketEnd(bizEntity.getPledgeAmountEnd());
    
    subBill.setAvailableAmount(BigDecimal.ZERO);  # 初始可用金额为0
    subBill.setNewBillStatus("003"); # 电票票据状态 = 已收票
    subBill.setBillStatus("010"); # 已质押
    subBill.setStockFlag("N");
    subBill.setSplitBillFlag("Y");

    subBill.setBoeId(boeId);
    subBill.setUseLeId(billBusUseIdMap.get(billDto.getId()));
    saveDtoList.add(subBill);  

    // 2. 可用子票
    remaining = ConvertUtils.sourceToTarget(billDto, LdgBillAccountDTO.class);
    remaining.setId(UuidUtils.getUuid());
    remaining.setTicketAmount(billDto.getTicketAmount().subtract(bizEntity.getBusAmount()));
    remaining.setBillStatus("001"); # 已签收
    remaining.setNewBillStatus("003"); # 电票票据状态 = 已收票
    remaining.setCirculateFlag("02"); # 流通标志 = 可流通
    remaining.setSplitBillFlag("Y");
    
    remaining.setOriBillId(billDto.getId()); # 拆分母票信息
    remaining.setOriBillCode(billDto.getPostalOrderCode());

    // 例如原始票据[00001, 10000]， 用掉 10块[00001, 01000]， 此票起始编码[01000+1, 10000]
    remaining.setSubticketStart(bizEntity.getPledgeAmountEnd() + 1); # 子票起始编码 = 交易子票结束编码 + 1
    remaining.setSubticketEnd(billDto.getSubticketEnd());            # 子票结束编码 = 母票的子票结束编码（始终继承）
    remaining.setStockFlag("Y");
    
    remaining.setUseLeId(billBusUseIdMap.get(billDto.getId()));
    childTicketList.add(remaining); 
    

    
    // 3. 母票状态修改为“已拆分”，电票票据状态变为“已终止”，流通标识变为“已结束”，是否在库为“否”
    billDto.setBillStatus("021"); # 已拆分
    billDto.setNewBillStatus("005"); # 已终止
    billDto.setCirculateFlag("09"); # 已结束
    billDto.setStockFlag("N"); # 是否在库为“否”
    updateDtoList.add(billDto);

coreLdgClient.saveAndUpdateBill({saveDtoList: saveDtoList, updateDtoList: updateDtoList, childTicketList: childTicketList});    

// 为生成的子票生成内部可使用列表记录： saveDtoList
for line in saveDtoList:
  lineLeDTO.setId(UuidUtils.getUuid());
  lineLeDTO.setBillId(line.getId());
  lineLeDTO.setLeId(line.getUseLeId());
  lineLeDTO.setBoeNo(line.getBoeNo());
  lineLeDTO.setInTransitAmount(BigDecimal.ZERO);
  lineLeDTO.setOccupiedAmount(line.getTicketAmount());
  lineLeDTO.setAvailableAmount(BigDecimal.ZERO);
  lineLeDTO.setInPoolAmount(BigDecimal.ZERO);
