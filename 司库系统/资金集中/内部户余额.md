### 内部户|余额查询
- https://127.0.0.1/fmsInnerBalance/page
```json
{
    "page": 1,
    "limit": 10,
    "startRegisterDate": "2026-07-01",
    "endRegisterDate": "2026-07-29",
    "orgPermissionType": "1",
    "bankNumber": "HBZ2901101-01",
    "isCountStat": "N"
}
```

```sql
select a.*, ia.account_name, ia.bank_number, ia.le_id, ia.account_type
	, ia.overdraw_flag, ia.interest_flag, ia.account_state
from fms_inner_account ia
	   inner join fms_inner_balance a on ia.id = a.account_id
where 1 = 1
	and a.group_id = 'af4ce58b58c721b9dcde3137322e0004'
	and a.enabled_flag = '0'
	and a.register_date >= TIMESTAMP '2026-07-01 00:00:00.000'
	and a.register_date < TIMESTAMP '2026-07-30 00:00:00.000'
	and ia.bank_number like '%HBZ2901101-01%'
	and (ia.cancel_date is null or a.register_date <= ia.cancel_date)
order by ia.bank_number, a.register_date desc
limit 21
```

```sql
select b.ID, a.BANK_NUMBER, a.ACCOUNT_NAME, a.LE_ID, b.COLLECT_TYPE
	, b.inner_account_id, b.CURRENCY_CODE, b.OPPOSITE_BANK_NUMBER, b.OPPOSITE_BANK_NAME, b.INTERIOR_DETAIL_STATUS
	, b.GENERATE_NODE, b.BUSINESS_DATE, b.REMARK, b.DEBIT_AMOUNT, b.CREDIT_AMOUNT
	, b.BUSINESS_BOE_NO, b.GROUP_ID, b.LAST_UPDATE_DATE, b.IS_SYNC, b.TRANSACTION_ID
	, b.DEBIT_FLAG, b.TRANS_BALANCE, b.TRANSACTION_PURPOSE, b.SOURCE, b.CREATE_BY
	, b.CREATE_DATE, b.BOE_NO, b.EXTERNAL_SYSTEM_KEY, a.CONTACT_USER_ID, a.ORG_ID
	, b.SOURCE_SYSTEM, b.BOE_TYPE_CODE, b.TASK_ID, b.BUSINESS_TYPE, b.FORM_TYPE_ID
	, b.MAPPED_TYPE, b.NEW_BILL_STATUS, b.BILL_TYPE_CODE, b.BUS_ID, b.BUS_DETAIL_ID
	, b.INTEREST_TAX_RATE, b.INTEREST_TAX_AMOUNT, b.INTEREST_EXCLUSIVE_AMOUNT, a.BANK_NUMBER as INNER_BANK_NUMBER,  a.ACCOUNT_NAME as INNER_ACCOUNT_NAME
	, b.UNIQUE_RECONCILIATION_CODE
	, case 
		when b.MATCH_STATUS is not null then b.MATCH_STATUS
		else '0'
	end as MATCH_STATUS, b.UNIQUE_NO
from FMS_INNER_ACCOUNT a
	inner join FMS_INTERIOR_DETAIL b on a.ID = b.INNER_ACCOUNT_ID
where 1 = 1
	and b.debit_amount != 0
	and b.debit_amount is not null
	and b.inner_account_id = '0d40dbb43977d3c73e94b1b7b2ae5b7e'
	and b.business_date >= TIMESTAMP '2026-07-23 00:00:00.000'
	and b.business_date <= TIMESTAMP '2026-07-23 23:59:59.999'
	and (b.CURRENCY_CODE = 'CNY')
	and b.INTERIOR_DETAIL_STATUS not in ('1', '2')
	and b.ENABLED_FLAG = '0'
	and b.GROUP_ID = 'af4ce58b58c721b9dcde3137322e0004'
order by BANK_NUMBER asc, business_date asc, last_update_date asc, id asc
limit 20
```
