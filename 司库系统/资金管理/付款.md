#### 生成支付指令
- cn.ztessc.service.payment.FmsPaymentInstructionsService#buildPayDataWhenPayGen
```python
    FmsPaymentInstructionsEntity fmsPaymentInstructionsEntity = ObjUtil.newInstance(FmsPaymentInstructionsEntity.class);
    //支付指令大部分字段来自付款申请单且字段相同，直接复制
    fmsPaymentInstructionsEntity = ConvertUtils.sourceToTarget(boeHeader, FmsPaymentInstructionsEntity.class);

    //设置指令编号
    String instructionNo = coreCommonClient.generateCode(RULE_INSTRUCTION_NO, boeHeader.getGroupId(), "");// 获取指令编号
    fmsPaymentInstructionsEntity.setInstructionNo(instructionNo.split("-")[1]);

    //设置单据ID
    fmsPaymentInstructionsEntity.setBoeId(boeHeader.getId());
    fmsPaymentInstructionsEntity.setLineId(boeHeader.getId());
    fmsPaymentInstructionsEntity.setSourceBoeNo(boeHeader.getSourceBoeNo());
    fmsPaymentInstructionsEntity.setPayerLeId(boeHeader.getPayLeId());
    fmsPaymentInstructionsEntity.setRelaBillNo(boeHeader.getPostalOrderCodes());
    fmsPaymentInstructionsEntity.setRelaBillId(boeHeader.getBillNumber());
    // 设置流程结束审批人
    fmsPaymentInstructionsEntity.setLastApproveCode(boeHeader.getLastApproveCode());
    // set新加字段:是否落地、是否跨行、大小额支付系统、同城异地
    fmsPaymentInstructionsEntity.setWhetherLanding(boeHeader.getWhetherLanding());
    fmsPaymentInstructionsEntity.setWhetherBank(boeHeader.getWhetherBank());
    fmsPaymentInstructionsEntity.setBigSmallPay(boeHeader.getBigSmallPay());
    fmsPaymentInstructionsEntity.setCityPlaces(boeHeader.getCityPlaces());
    fmsPaymentInstructionsEntity.setOldPaymentCode(boeHeader.getOldPaymentCode());
    fmsPaymentInstructionsEntity.setOldPaymentName(boeHeader.getOldPaymentName());

    //默认待支付
    fmsPaymentInstructionsEntity.setPaymentStatus("10");
    if (StrUtil.equals(boeHeader.getPaymentModeCode(), "13")) {
        // 结算方式：内部转账  付款申请单审批完成修改付款状态为已付款
        fmsPaymentInstructionsEntity.setPaymentStatus("30");
    }     
    //默认待发送
    fmsPaymentInstructionsEntity.setSendStatus("1");

    
    //默认重发次数0
    fmsPaymentInstructionsEntity.setRepeatSendCount(0);
    if (StrUtil.isNotBlank(boeHeader.getRepeatBoeNo())) {        
        // 重付申请单生成指令 重发次数1
        fmsPaymentInstructionsEntity.setRepeatSendCount(1);
    }
    
    //默认指令支付状态查询次数0
    fmsPaymentInstructionsEntity.setQueryCount(0);
    fmsPaymentInstructionsEntity.setExtendNum(BigDecimal.ZERO);
    
    FmsAccountEntity fmsAccountEntity = fmsAccountService.findOne(boeHeader.getPayerBankAccountId());
    // 根据支付方式以及账户信息判断指令类型
    if (FmsEnumType.FMS_PAYMENT_MODE.PAYMENT_MODE_1.getValue().equals(boeHeader.getPaymentModeCode())) {
        //判断付款账户网银直联是否开通,未开通时付款状态置为失败
        if (!FmsEnumType.E_BANK_MONOBLOCK.E_BANK_MONOBLOCK_1.getValue()
                .equals(fmsAccountEntity.geteBankMonoblock())) {
            fmsPaymentInstructionsEntity.setPaymentStatus(
                    FmsEnumType.FMS_PAYMENTS_STATUS.PAYMENTS_STATUS_24.getValue());
            fmsPaymentInstructionsEntity.setFailReason(MessageUtils.getMessage(29050));
        }
        //支付方式为网银转账时，指令类型为转账指令
        fmsPaymentInstructionsEntity.setInstructionType(FmsEnumType.INSTRUCTION_TYPE.INSTRUCTION_TYPE_1.getValue());
    } else {
        //其他支付方式进入线下支付
        fmsPaymentInstructionsEntity.setInstructionType(FmsEnumType.INSTRUCTION_TYPE.INSTRUCTION_TYPE_3.getValue());
    }
    
    // 设置CA标识
    if (StrUtil.isNotBlank(boeHeader.getCaSignedData())) {
        // 生成指令的时候不给密文。
        fmsPaymentInstructionsEntity.setCaPlainData(null);
        fmsPaymentInstructionsEntity.setCaSignedData(null);
    }
    // 根据CA预写标识 打上CA预写标识
    String preCaFlag = boeHeader.getPreCaFlag();
    fmsPaymentInstructionsEntity.setPreCaFlag(preCaFlag);
    
    if (FmsEnumType.FMS_BOE_TYPE.PAYMENT_BOE.getValue().equals(boeHeader.getBoeTypeCode())) {
        fmsPaymentInstructionsEntity.setBusinessType(FmsEnumType.BUSINESS_TYPE.PAYMENT_REQUISITION_BOE.getValue());
    } else if (FmsEnumType.FMS_BOE_TYPE.BATCH_PAYMENT_BOE.getValue().equals(boeHeader.getBoeTypeCode())) {
        fmsPaymentInstructionsEntity.setBusinessType(FmsEnumType.BUSINESS_TYPE.BATCH_PAYMENT_BOE.getValue());
    }
    // 结算方式为联动支付时，付款审批完成后进入指令发送状态，一笔挂起状态的付款指令
    String settlementType = boeHeader.getSettlementType();
    if (StrUtil.equals(settlementType, FmsEnumType.SettlementType.SETTLEMENT_TYPE_2.getValue())) {
        fmsPaymentInstructionsEntity.setHangStatus(FmsEnumType.HANG_STATUS.HANG_STATUS_2.getValue());
    }
    fmsPaymentInstructionsEntity.setReviewStatus(FmsEnumType.InsReviewStatus.INS_REVIEW_STATUS_O1.getValue());
    EntityUtils.setWho(fmsPaymentInstructionsEntity);
```
