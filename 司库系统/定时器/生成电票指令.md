# 生成电票指令


### cn.ztessc.service.task.JobCenter#generateBillInstructions
```sql

select * 
  from zfs_fms.FMS_BOE_HEADER
 where BOE_STATUS = '30'
   and VALIDITY_FLAG = '0'
   and BOE_TYPE_CODE = 'BILL_BOE'
   and DISPOSE_TYPE = '0' -- 处理方式 0：直联 1：线下
   and INSTRUCTION_FLAG is null -- 指令生成是否成功 0：成功 1：失败


select * 
  from zfs_fms.fms_boe_bill_receive_busi
 where BOE_ID in (:boeIds)
   and VALIDITY_FLAG = '0'


select * 
  from zfs_fms.FMS_BOE_BILL_APPLY
 where BOE_ID in (:boeIds)
   and VALIDITY_FLAG = '0'

-- 查询票据id
select BILL_ID 
  from zfs_fms.FMS_BOE_BILL_RECEIVE
 where BOE_ID in (:boeIds)
   and VALIDITY_FLAG = '0'
```
