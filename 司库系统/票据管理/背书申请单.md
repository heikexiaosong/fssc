# 背书申请单
- cn.ztessc.service.FmsEndorseApplyBoeService

票据业务单-应收：
FMS_BOE_BILL_RECEIVE_BUSI：【BOE_ID = boeHeader.getId()】

票据业务表：
FMS_BILL_BUSINESS：【BOE_ID = boeHeader.getId() and BILL_ID in (:billIds) 】


### 单据提交
- cn.ztessc.service.FmsEndorseApplyBoeService#doSubmit
```sql

-- bodId: boeBillPaymentApply[0]::id 关联付款申请
select * from FMS_PAYMENT_APPLY_BILL where BOE_ID = :bodId

-- occupyBills: [boeBillReceiveDTOs::billId]; 获取已经占用的票据Id, 从单据的【票据业务单-关联票据区】获取
-- currentBillIds: [FMS_PAYMENT_APPLY_BILL::BILL_ID]; 获取当前背书申请单关联的所有票据Id


/**
  *  批量写入付款申请单ID
  */
@PostMapping("ldgbillaccount/updatePayBoeIdByIdList")
Result updatePayBoeIdByIdList(@RequestBody Map<String, Object> params);

```

### 单据审批完成
- cn.ztessc.service.FmsEndorseApplyBoeService#workFlowEndEvent

```sql

-- fmsBoeBillReceiveBusi
select * from FMS_BOE_BILL_RECEIVE_BUSI where BOE_ID = :boeId

-- fmsBoeBillReceiveDTOs
select * from FMS_BOE_BILL_RECEIVE where BOE_ID = :boeId

  -- boePaymentApplyList: 查询背书申请单关联的付款申请单列表
  -- paymentApplyId = boePaymentApplyList[0]::applyBoeId
  select * from FMS_BOE_PAYMENT_APPLY where BOE_ID = :boeId
  
  -- paymentApplyBills: 查询付款申请单关联的所有票据
  select * from FMS_PAYMENT_APPLY_BILL where BOE_ID = :paymentApplyId
  
  -- busAmount：业务金额（背书）； billId：票据ID
  -- occupyAmountMap: paymentApplyBills<billId, busAmount[累加和]>

```
