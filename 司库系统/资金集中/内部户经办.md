# 内部户单据经办

### 获取经办信息
- 【FMS】 https://127.0.0.1/fmsInnerAccountHandle/queryAccountInfo
- cn.ztessc.controller.inneraccount.FmsInnerAccountHandleController#queryAccountInfo
```json
{
    "boeNo": "HUASHI-FX2608120032",
    "boeTypeCode": "NBHXHSQD"
}
```

```python
def queryAccountInfo(params):
    boeNo = params.get("boeNo")
    # 根据单据编号查询账户经办数据 FMS_INNER_ACCOUNT_HANDLE[BOE_NO=:boeNo]
    fmsInnerAccountHandleDto = this.findByBoeNo(boeNo)
    if not fmsInnerAccountHandleDto:
        return ObjUtil.newInstance(FmsInnerAccountDTO)
        
    boeFormDto = zfsFmsAccountInnerDaoFoeClient.findById(fmsInnerAccountHandleDto.getBoeHeaderId())
    accountInfo = getBoeLine("accountInfo", boeFormDto.getZfsBoeLines())
    
    # 账户开户申请单：填充单据详情信息
    fmsInnerAccountDto =  ObjUtil.newInstance(FmsInnerAccountDTO.class);
    fmsInnerAccountDto.setLeId(fmsInnerAccountHandleDto.getLeId());
    fmsInnerAccountDto.setAccountName(fmsInnerAccountHandleDto.getAccountName());
    fmsInnerAccountDto.setAccountType(fmsInnerAccountHandleDto.getAccountType());
    fmsInnerAccountDto.setAccountCurrencyCodes(fmsInnerAccountHandleDto.getAccountCurrencyCodes());
    fmsInnerAccountDto.setAccountCurrencyCodeNames(fmsInnerAccountHandleDto.getAccountCurrencyCodeNames());
    //数据来源 1-开户申请
    fmsInnerAccountDto.setDataSource(FmsEnumType.ACCOUNT_DATA_SOURCE.OPEN_APPLICATION.getCode());
    //翻译名称
    baseDataTranslateUtils.baseDataTranslate(fmsInnerAccountDto);
    
    # 处理附件信息
    fmsAttachmentDtoList = handleAttach(fmsInnerAccountHandleDto)
    fmsInnerAccountDto.setAttachmentList(fmsAttachmentDtoList)    
    
    result = DtoMapConverter.convertDtoToMapAuto(fmsInnerAccountDto)
    
    if accountInfo:
        result.update(DtoMapConverter.convertDtoToMapAuto(accountInfo))
        result.update(accountInfo.getExtendMap())
        
    return result;
```
