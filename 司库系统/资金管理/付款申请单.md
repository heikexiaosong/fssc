```python
# 原始单据 FmsPaymentApplyHeaderEntity headerEntity
headerEntity = getAndLockOriginalHeader(splitPaymentApplyHeader.getBoeId());

# 关联单据 List<FmsPaymentApplyRelaBoeDTO> relaBoeList
relaBoeList = fmsPaymentApplyRelaBoeService.findByHeaderId(splitPaymentApplyHeader.getBoeId());

# FmsPaymentApplyHeaderDTO dto
for dto in splitPaymentApplyHeader.getSplitList():
    entity = ConvertUtils.sourceToTarget(headerEntity, FmsPaymentApplyHeaderEntity.class);
    entity.setId(UuidUtils.getUuid());
    entity.setBoeNo(generateCode("PAYMENT_BOE_NO"));
    entity.setBoeStatus(FmsEnumType.FMS_BOE_STATUS.BOE_STATUS_10.getValue());
    entity.setEmployeeId(UserUtils.getEmpId());
    
    entity.setPaymentModeCode(dto.getPaymentModeCode());
    entity.setTradPurpose(dto.getTradPurpose());
    entity.setBoeAbstract(dto.getBoeAbstract());
    entity.setApplyAmount( dto.getApplyAmount());   
    # 设置剩余可付款金额
    entity.setResidueAmount(entity.getApplyAmount()); 
    
    # 付款账号信息
    FmsAccountDTO accountDto = fmsAccountService.findById(dto.getPayerBankAccountId());
    entity.setPayerBankAccountId(accountDto.getId());
    entity.setPayerBankAccountNum(accountDto.getBankNumber());
    entity.setPayerBankAccountName(accountDto.getAccountName());
    entity.setPayerBankHeadCode(accountDto.getBankHeadId());
    entity.setPayerBankHeadName(accountDto.getBankHeadIdName());
    entity.setPayerBankBranchName(accountDto.getOpenBankName());
    entity.setPayerBankArea(accountDto.getAreaCode());
    entity.setPayerUnitedCode(accountDto.getUnitedCode());
    
    entity.setFirstBoeDate(null);
    entity.setProcessInstanceId(null);
    entity.setNodeName(null);
    entity.setWfTaskId(null);
    entity.setOldInstructionNo(null);
    # 0: 拆分、1: 合并、 2: 代发动作标识.
    entity.setActionFlag(Constant.DEFAULT_FLAG_YES);
    entity.setRelatedBoeNo(headerEntity.getBoeNo());    
    # 重新生成 externalSystemKey
    entity.setExternalSystemKey(UUID.randomUUID().toString());

    splitEntityList.add(entity);
    if (CollUtil.isNotEmpty(relaBoeList)) {
        relaBoeEntities = ConvertUtils.sourceToTarget(relaBoeList, FmsPaymentApplyRelaBoeEntity.class);
        for (FmsPaymentApplyRelaBoeEntity relaBoeEntity : relaBoeEntities) {
            relaBoeEntity.setId(UuidUtils.getUuid());
            relaBoeEntity.setBoeId(entity.getId());
            relaBoeEntity.setCreateBy(userId);
            relaBoeEntity.setCreateDate(new Date());
        }
        relaBoeEntityList.addAll(relaBoeEntities);
    }
```
