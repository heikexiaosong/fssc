            elif valueForm == "assignVariable":
                // 指定变量，原样返回，交给实现端自己找，赋值科目指定变量
                linesResDTO.setSubjectAssignVariable(value);
                
            elif valueForm == "relationScheme":
                // 关联方案 customizeAccountsId = value 科目方案ID
                
            

        
        // 解析原币金额设置
        parseOriginAmount(templateLinesDTO.getOrignAmountConfig(), data, linesResDTO);
        // 解析原币币种设置
        this.parseOriginCurrencyConfig(templateLinesDTO.getOrignCurrecyConfig(), data, linesResDTO, paramMap);
        // 解析汇率配置
        this.parseRateConfig(templateLinesDTO.getRateConfig(), data, linesResDTO);
        // 解析行摘要设置

        // 解析分录金额配置
        this.parseEntryAmountVariable(templateLinesDTO.getEntryAmountConfig(), data, linesResDTO);
        
        linesResDtoList.add(linesResDTO);   
    }
