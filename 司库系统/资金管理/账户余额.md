### 【定时器】当日余额同步到历史余额
- synManualBalanceDayToBalanceHis



INSERT INTO fms_balance_his (
    ID, VALIDITY_FLAG, ENABLED_FLAG, CREATE_BY, CREATE_DATE, 
    LAST_UPDATE_BY, LAST_UPDATE_DATE, GROUP_ID, ARCHIVE_FLAG, ARCHIVE_DATE, 
    EXTEND_S1, EXTEND_N1, EXTEND_D1, ACCOUNT_ID, ACCOUNT_CODE, 
    BANK_NUMBER, ACCOUNT_NAME, LE_ID, OPEN_BANK, BANK_HEAD_CODE, 
    ACCOUNT_TYPE, CURRENCY_CODE, E_BANK_MONOBLOCK, REGISTER_DATE, DATA_SOURCES, 
    SYNC_DATE, 
    REGISTER_DATE_NUM,   -- 特有字段
    OPEN_BANK_NAME, 
    REGISTER_DATE_STR,   -- 特有字段
    AVAILABLE_BALANCE, BOOK_BALANCE, POOL_BALANCE, USEABLEQUOTA, AVAOVERBAL, 
    EXTEND_BALANCE1, EXTEND_BALANCE2, EXTEND_BALANCE3, EXTEND_ST1, EXTEND_ST2, 
    EXTEND_ST3, PROD_11, PROD_12, PROD_13, BANK_HEAD_NAME, EXCHANGE_RATE, 
    ACCOUNT_BALANCE_RMB, -- 对应源表 BOOK_BALANCE_RMB
    AVAILABLE_BALANCE_RMB, DEPOSIT_BALANCE_RMB, ONSHORE
)
SELECT 
    ID, -- 生成新的唯一ID
    VALIDITY_FLAG, ENABLED_FLAG, CREATE_BY, NOW(), 
    LAST_UPDATE_BY, LAST_UPDATE_DATE, GROUP_ID, '0', NOW(), -- 标记为已归档
    EXTEND_S1, EXTEND_N1, EXTEND_D1, ACCOUNT_ID, ACCOUNT_CODE, 
    BANK_NUMBER, ACCOUNT_NAME, LE_ID, OPEN_BANK, BANK_HEAD_CODE, 
    ACCOUNT_TYPE, CURRENCY_CODE, E_BANK_MONOBLOCK, REGISTER_DATE, DATA_SOURCES, 
    SYNC_DATE, 
    CAST(DATE_FORMAT(REGISTER_DATE, '%Y%m%d') AS UNSIGNED), -- 转换为 20231027
    OPEN_BANK_NAME, 
    DATE_FORMAT(REGISTER_DATE, '%Y-%m-%d'), -- 转换为 '2023-10-27'
    AVAILABLE_BALANCE, BOOK_BALANCE, POOL_BALANCE, USEABLEQUOTA, AVAOVERBAL, 
    EXTEND_BALANCE1, EXTEND_BALANCE2, EXTEND_BALANCE3, EXTEND_ST1, EXTEND_ST2, 
    EXTEND_ST3, PROD_11, PROD_12, PROD_13, BANK_HEAD_NAME, EXCHANGE_RATE, 
    BOOK_BALANCE_RMB, -- 映射
    AVAILABLE_BALANCE_RMB, DEPOSIT_BALANCE_RMB, ONSHORE
FROM fms_balance_day
WHERE REGISTER_DATE >= '2026-07-10 00:00:00' 
  AND REGISTER_DATE <= '2026-07-10 23:59:59'
  AND ENABLED_FLAG = '0';
