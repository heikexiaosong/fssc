### 同步资金系统当日流水
- cn.ztessc.job.JobCenter#synFmsTransactionDetailToday

@PostMapping("apiController/transDetailDay")
Result transDetailDay(@RequestParam("param") String param);

```python
param = {
    "page": 1,
    "limit": 1000,
    "payOnlyDeductFlag": "Y",
    "beginLastUpdateDate": "yyyy-MM-dd HH:mm:ss",
    "endLastUpdateDate": "yyyy-MM-dd HH:mm:ss"
}
result = zfsClaimServiceCoreFmsClient.transDetailDay(param);
# 交易明细接口返回参数【result】解析，写入CLAIM_FLOW_SUMMARY表
jsonArray = result["data"]["list"]
for item : jsonArray:
    dto = ObjUtil.newInstance(ClaimFlowSummaryDTO.class);
    dto.setId(item.getString("id"));
    # 交易流水号
    dto.setHostid(item.getString("transactionNo"));
    dto.setClaimBusinessNo(item.getString("fmsBusinessNo"));  # 资金系统内生成的唯一KEY
    dto.setUniqueNo(itemJson.getString("uniqueNo")); #  要素10 特殊唯一要素
    
    # 交易用途
    dto.setUseage(item.getString("transactionPurpose"));
    dto.setActDate(item.getString("transactionDate"));
    dto.setExternalId(item.getString("id"));  # 外部系统唯一ID 设置为资金系统的ID
    
    # 回单url
    dto.setReceiptUrl(item.getString("billReceiptUrl"));    
    dto.setReceiptAccessoryName(item.getString("billReceiptNo")); # 回单名称
    dto.setReceiptSource("1"); # 如果回单地址不为空 设置回单附件来源为接口同步
    
    dto.setImportType("api"); # 接口导入    
    dto.setClaimStatus("10"); # 待清分

    # 回单 url/名称 > 60 字符	跳过	超长不同步
    
    

```
