-- 2. 查询原授信 有已占用额度的授信品种
select fpr.CREDIT_BREED_TYPE, fpr.SUB_CREDIT_BREED_TYPE, fpr.*  from FMS_CREDIT_BREED_CHILD fpr

  and ENABLED_FLAG  = '0';;

SELECT START_DATE, END_DATE FROM FMS_BREED_CHILD_USE t WHERE t.BREED_ID IN (
 SELECT temp.BREED_ID FROM (  SELECT  used.BREED_ID,sum(used.AMOUNT) 
   FROM FMS_BREED_CHILD_USE used  
  WHERE used.CREDIT_ID = 'd25dcf05-154c-41b2-ac0e-90e2978bcbba'  and used.VALIDITY_FLAG = '0' 
  GROUP BY used.BREED_ID HAVING SUM(used.AMOUNT)>0  ) temp
