### 内部借款到期自动扣款
- 【内部借款过期定时器】 syncAutoInnerLoanLedger
- 内部借款到期自动扣款自动生成还款登记记录
```sql
-- 
select *
  from fms_inner_loan_ledger fmsinnerlo0_ 
 where fmsinnerlo0_.deduc_auto_flag = 'Y'   -- 是否到期自动扣款
   and fmsinnerlo0_.loan_end_date <= TIMESTAMP '2026-09-07 00:00:00.000' 
   and fmsinnerlo0_.loan_status not in ('5', '0')  -- 业务状态 1-已登记、2-已放款、3-逾期、4-展期、5-结清、0-已废弃
   and fmsinnerlo0_.enabled_flag = '0' 
   and fmsinnerlo0_.validity_flag = '0' 
   and fmsinnerlo0_.group_id = 'af4ce58b58c721b9dcde3137322e0004'

-- 获取放款记录列表：list
select t.* 
  from FMS_INNER_LOAN_LEDGER_RECORD t 
 where 1 = 1 
   and t.loan_Id in ('8684d1e4270b11a471bc3137322e6f8f', '2df445a24fdf47f508d13137322e6f13') 
   and t.group_id = 'af4ce58b58c721b9dcde3137322e0004' 
   and t.type = '1'  -- 类型 0-利率调整，1-放款，2-还款，3-展期
   and t.enabled_flag = '0' 
order by t.effective_date asc
```

```sql
select r.*, i.loan_code, i.loan_status, i.lender_le_id, i.loaner_le_id
     , i.loan_way, ir.interest_way, i.loan_begin_date, i.loan_end_date, ir.billing_month
     , ir.billing_day, i.lender_inner_id, i.loaner_inner_id, ir.disbursement_detail_code, ir.last_interest_date
     , ir.last_withhold_date, i.credit_amount_sum, i.repay_amount_sum, ir.first_interest_date_type, ir.first_interest_custom_date
     , ir.effective_date
from FMS_INNER_LOAN_INTEREST_RECORD r
     inner join FMS_INNER_LOAN_LEDGER i on r.LOAN_ID = i.id
     inner join FMS_INNER_LOAN_LEDGER_RECORD ir on ir.ID = r.LOAN_RECORD_ID and ir.enabled_flag = '0'
where i.enabled_flag = '0'
  and r.enabled_flag = '0'
  and r.validity_flag = '0'
  and r.rate != 0
  and i.group_id = '98384a38c6501ea777273132372e0008'
  and i.loan_code like '%NBJK2611110003%'
  and i.LOAN_STATUS in ('2', '3', '4')
  and i.LOAN_WAY not in ('9', '10')
  and r.BEGIN_DATE < TIMESTAMP '2026-11-12 00:00:00.000'
  and r.END_DATE > ir.MAX_INTEREST_DATE
  
  and ir.IS_INTEREST = '1'
  and ir.INTEREST_WAY in ('1', '04', '05', '06', '07')
  and ir.MAX_INTEREST_DATE < TIMESTAMP '2026-11-12 00:00:00.000'
  and (i.LOAN_END_DATE <= TIMESTAMP '2026-11-12 00:00:00.000' or (ir.BILLING_MONTH like '%11%' and ir.BILLING_DAY = '12'))
order by i.loan_code asc, ir.disbursement_detail_code asc, r.interest_type asc, r.begin_date asc
```
