# 同步当日余额数据
- synBalanceDay


```python
//获取所有未删除、未禁用、未销户、直联开通、余额查询开通的账户
Map<String, Object> params = new HashMap<>(5);
params.put(E_BANK_MONOBLOCK, Constant.DEFAULT_FLAG_YES);
params.put("balanceInquiry", Constant.DEFAULT_FLAG_YES);
params.put("salesMarkNe", FmsConstants.salesMark.SALES.getCode());
String jobParams = jobParamDto.getParams();
if (StrUtil.isNotBlank(jobParams)) {
    List<String> paramsBankCodeList = StrUtil.split(jobParams, Constant.SEPARATOR_COMMA, 0, true, true);
    params.put("paramsBankCodeList", paramsBankCodeList);
}
Long accountCount = fmsAccountService.findCountBySql(new Query(params));
```

```sql
-- 查询待同步账号
select *
  from fms_account fav 
 where 1 = 1 
  and fav.E_BANK_MONOBLOCK = '0' 
  and fav.BALANCE_INQUIRY = '0' 
  and fav.SALES_MARK <> '1' 
  and fav.enabled_flag = '0' 
  and fav.validity_flag = '0' 
  and fav.BANK_HEAD_ID in ('102')

-- 币种信息
select *
  from fms_account_currency fmsaccount0_
 where fmsaccount0_.account_id in ('a9262d0e688ea82460d2b1b0aeb55b5b', 'a9262d0e688ea81c9d99b1b0aeb55b57')
   and fmsaccount0_.enabled_flag = '0'
   and fmsaccount0_.group_id = 'af4ce58b58c721b9dcde3137322e0004'

-- 
select *
  from fms_currycode_relation fmscurrenc0_
 where fmscurrenc0_.bank_head_code = '102'
   and fmscurrenc0_.enabled_flag = '0'
 order by fmscurrenc0_.create_date desc

select *
  from fms_account_sign_online fmsaccount0_
 where fmsaccount0_.account_id in ('a9262d0e688ea82460d2b1b0aeb55b5b', 'a9262d0e688ea81c9d99b1b0aeb55b57')
   and fmsaccount0_.enabled_flag = '0'
   and fmsaccount0_.group_id = 'af4ce58b58c721b9dcde3137322e0004'

select *
  from fms_access_institution fmsaccessi0_
 where fmsaccessi0_.bank_head_id in ('102')
   and fmsaccessi0_.enabled_flag = '0'
   and fmsaccessi0_.group_id = 'accba517-5725-4fae-8938-48946676bfc1'

select *
  from fms_params fmsparamse0_
 where fmsparamse0_.param_code = 'SAME_GROUP_COMPANY_SEPARATE_USB_KEY'
   and fmsparamse0_.group_id = 'accba517-5725-4fae-8938-48946676bfc1'
   and fmsparamse0_.enabled_flag = '0'
   and fmsparamse0_.validity_flag = '0'
limit 2
```
