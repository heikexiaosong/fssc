### 【定时器】银企对账统计数据
- cn.ztessc.service.task.JobCenter#reconciliationStatisticsData
```python
# sql: select count(1) from fms_reconciliation_account fav where 1=1
Long accountCount = fmsReconciliationAccountService.findCountBySql(new Query({}));

# 分页查询sql: select * from fms_reconciliation_account fav where 1=1
List<FmsReconciliationAccountEntity> fmsReconciliationAccountList = fmsReconciliationAccountService.queryPageNoCountBySql(
                    new Query(params));

# key: bankNumber^transactionCurrencyCode
# bank: FROM FMS_TRANSACTION_DETAILS_HIS WHERE ENABLED_FLAG='0' AND RECONCILIATION_STATUS IN ('1','4','5') AND BANK_NUMBER = :bankNumber 
Map<String, List<FmsTransactionDetailsHisDTO>> bankBankNumberMap = new HashMap<>(fmsReconciliationAccountList.size());
#  erp: FROM FMS_ERP_TRANSACTION         WHERE ENABLED_FLAG='0' AND RECONCILIATION_STATUS IN ('1','4','5') AND BANK_NUMBER = :bankNumber 
Map<String, List<FmsErpTransactionDTO>>        erpBankNumberMap  = new HashMap<>(fmsReconciliationAccountList.size());


# 银企对账工作台记录
List<FmsReconciliationWorkbenchEntity> workbenchList = new ArrayList<>();

List<String> accountIdList = new ArrayList<>(fmsReconciliationAccountList.size());
for entity:FmsReconciliationAccountEntity in fmsReconciliationAccountList:
    accountIdList.add(entity.getId());
    
    String bankNumber = entity.getBankNumber();
    String currency = entity.getCurrency();
    String key = bankNumber + Constant.DELIMITER_ANGLE_ESCAPE + currency;

    List<FmsTransactionDetailsHisDTO> hisAccList = bankBankNumberMap.get(key);
    hisTranMap = hisAccList<transactionDate{yyyy-MM-dd}, []>

    List<FmsErpTransactionDTO> erpAccList = erpBankNumberMap.get(key);
    erpTranMap = erpAccList<transactionDate{yyyy-MM-dd}, []>

    for dateStr, hisList in hisTranMap.items():
        hisAccMap = hisList<reconciliationStatus, {}>
        
        erpList = erpTranMap.get(dateStr);
        erpAccMap = erpList<reconciliationStatus, {}>

        FmsReconciliationWorkbenchEntity workbenchEntity = ObjUtil.newInstance(FmsReconciliationWorkbenchEntity.class);
        workbenchEntity.setId(UuidUtils.getUuid());
        workbenchEntity.setAccountId(entity.getId());
        workbenchEntity.setGroupId(entity.getGroupId());
        workbenchEntity.setRegisterDate(DateUtil.parse(dateStr, DatePattern.NORM_DATE_PATTERN));
        # 组装历史交易数据
        for reconciliationStatus, hisDto in hisAccMap.items():
            Integer    number               = hisDto.getMatchTime();
            BigDecimal transactionAmount    = hisDto.getTransactionAmount();
            
            if reconciliationStatus == '1':
                # RECONCILIATION_STATUS_1("1", "未对账")
                workbenchEntity.setUnreconBankNum(number);
                workbenchEntity.setUnreconBankAmount(transactionAmount);
            if reconciliationStatus == '5':
                # RECONCILIATION_STATUS_5("5", "未达账")
                workbenchEntity.setUnaccountsBankNum(number);
                workbenchEntity.setUnaccountsBankAmount(transactionAmount);
            if reconciliationStatus == '4':
                # RECONCILIATION_STATUS_4("4", "已对账")
                workbenchEntity.setReconedBankNum(number);
                workbenchEntity.setReconedBankAmount(transactionAmount);
            
            # 获取企业账数据
            FmsErpTransactionDTO erpDto = erpAccMap.get(reconciliationStatus);
            if (ObjUtil.isEmpty(erpDto)) {
                // 无企业账数据
                workbenchEntity.setReconedErpAmount(BigDecimal.ZERO);
                continue;
            }
            
            number              = erpDto.getMatchTime();
            transactionAmount   = erpDto.getTransactionAmount();
            if reconciliationStatus == '1':
                # RECONCILIATION_STATUS_1("1", "未对账")
                workbenchEntity.setUnreconErpNum(number);
                workbenchEntity.setUnreconErpAmount(transactionAmount);
            if reconciliationStatus == '5':
                # RECONCILIATION_STATUS_5("5", "未达账")
                workbenchEntity.setUnaccountsErpNum(number);
                workbenchEntity.setUnaccountsErpAmount(transactionAmount);
            if reconciliationStatus == '4':
                # RECONCILIATION_STATUS_4("4", "已对账")
                workbenchEntity.setReconedErpNum(number);
                workbenchEntity.setReconedErpAmount(transactionAmount);
        
        if (ObjUtil.isEmpty(workbenchEntity.getUnreconBankNum())) {
            workbenchEntity.setUnreconBankNum(FmsConstants.FMS_ZERO);
            workbenchEntity.setUnreconBankAmount(BigDecimal.ZERO);
        }
        if (ObjUtil.isEmpty(workbenchEntity.getUnaccountsBankNum())) {
            workbenchEntity.setUnaccountsBankNum(FmsConstants.FMS_ZERO);
            workbenchEntity.setUnaccountsBankAmount(BigDecimal.ZERO);
        }
        if (ObjUtil.isEmpty(workbenchEntity.getReconedBankNum())) {
            workbenchEntity.setReconedBankNum(FmsConstants.FMS_ZERO);
            workbenchEntity.setReconedBankAmount(BigDecimal.ZERO);
        }
        if (ObjUtil.isEmpty(workbenchEntity.getUnreconErpNum())) {
            workbenchEntity.setUnreconErpNum(FmsConstants.FMS_ZERO);
            workbenchEntity.setUnreconErpAmount(BigDecimal.ZERO);
        }
        if (ObjUtil.isEmpty(workbenchEntity.getUnaccountsErpNum())) {
            workbenchEntity.setUnaccountsErpNum(FmsConstants.FMS_ZERO);
            workbenchEntity.setUnaccountsErpAmount(BigDecimal.ZERO);
        }
        if (ObjUtil.isEmpty(workbenchEntity.getReconedErpNum())) {
            workbenchEntity.setReconedErpNum(FmsConstants.FMS_ZERO);
            workbenchEntity.setReconedErpAmount(BigDecimal.ZERO);
        }
        
        workbenchList.add(workbenchEntity);
        # 删除企业账数据
        erpTranMap.remove(dateStr);       
    
# 先删除数据，再新增
# sql: delete from FMS_RECONCILIATION_WORKBENCH where ACCOUNT_ID in :accountIdList
fmsReconciliationWorkbenchDao.deleteByAccountIdList(accountIdList);
if (CollUtil.isNotEmpty(workbenchList)) {
    EntityUtils.setWho(workbenchList);
    this.insertBatch(workbenchList);
}


```
