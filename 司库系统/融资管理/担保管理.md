select *
  from fms_internal_guarantee 
 where (is_other_guarantees = '' or is_other_guarantees is null or is_other_guarantees = '')
   and warranty_number like '%727担保占用金额变更测试%' 
   and registrar_role = '2'
   and currency_code = 'CNY'
   and status = '1'
   and guarantor_id = '0b22741bcbaf4eb14b853139322e7d95'
   and enabled_flag = '0'
   and group_id = 'ba2ad44f9aba2ad54fa43139322e0000'
order by create_date desc, id desc
