# 共享查询资金付款申请单终态

### 【fscTask】queryEfsPayStatus

```java
/**
 * 指令状态查询(多条)
 *
 * @param param
 * @return
 */
@PostMapping("apiController/queryPaymentStatusByMore")
Result queryPaymentStatusByMore(@RequestBody String param);

param = [OpEfsPaymentDTO::id]


// 【fms】cn.ztessc.service.payment.FmsPaymentInstructionsService#queryPaymentStatusList(String[] externalSystemKeys)

// 指令信息 Map<String, List<FmsPaymentInstructionsEntity>> paymentInstructionMap
select * from FMS_PAYMENT_INSTRUCTIONS where EXTERNAL_SYSTEM_KEY in (:externalSystemKeys) and enabled_flag = '0';

// 付款申请单信息 Map<String, FmsPaymentApplyHeaderEntity> applyHeaderMap
select * from FMS_PAYMENT_APPLY_HEADER where EXTERNAL_SYSTEM_KEY in (:externalSystemKeys) and enabled_flag = '0';





```
