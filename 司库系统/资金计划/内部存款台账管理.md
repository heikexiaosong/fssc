--  查询台账信息
select * from FMS_INNER_DEPOSIT_LEDGER  where id = 'c22eae6b242f832ebb35b1b7b2ae15ea'

--  校验台账核算主体下的 活期账户 
select * 
from fms_inner_account fav 
where 1 = 1 
and fav.group_id = 'af4ce58b58c721b9dcde3137322e0004' 
and fav.le_id = '712eea2f6f8d723b207ab1b7b2ae16b1' 
and fav.validity_flag = '0' 
and fav.enabled_flag = '0' 
and fav.account_state = '0' 
and fav.account_type in ('01')   // 活期账户
and exists ( 
        select fac.currency_code 
                from fms_inner_currency fac 
        where fav.id = fac.account_id 
                        and fac.currency_code = 'CNY' 
        ) 
        and fav.id in ('f9f191860a6f926538f5b1b7b2ae1e3f') 
order by fav.create_date desc, fav.last_update_date desc 
limit 10

```sql
--
select * 
from fms_inner_deposit_ledger fmsinnerde0_
where fmsinnerde0_.id = '6c74cc915827d559e278b1b7b2ae3f48'

-- 
select *
  from FMS_INNER_DEPOSIT_FLOW_RECORD R
 where sort_num = (
            select max(sort_num)
              from FMS_INNER_DEPOSIT_FLOW_RECORD R
              where 1 = 1
                and R.DEPOSIT_ID = '6c74cc915827d559e278b1b7b2ae3f48'
                and R.ENABLED_FLAG = '0'
                and R.FLOW_TYPE = '3'
                and R.ROTATION_TYPE = 'LXzq')
  and R.DEPOSIT_ID = '6c74cc915827d559e278b1b7b2ae3f48'
  and R.ENABLED_FLAG = '0'
  and R.FLOW_TYPE = '3'
  and R.ROTATION_TYPE = 'LXzq'
limit 1

-- 
select *
  from FMS_INNER_INTERRATE_RECORD t
       inner join (
        select INNER_ID, max(effective_date) as effective_date
          from FMS_INNER_INTERRATE_RECORD t
         where 1 = 1
           and t.INNER_ID in ('468b2cbe105588061583b1b7b2ae47e4')
           and t.effective_date <= TIMESTAMP '2026-07-28 00:00:00.000'
           and t.enabled_flag = '0'
         group by t.INNER_ID
      ) tmp on t.INNER_ID = tmp.INNER_ID
  and t.effective_date = tmp.effective_date
  and t.INNER_ID in ('468b2cbe105588061583b1b7b2ae47e4')
  and t.enabled_flag = '0'
```
