### 内部借款单据经办
- 【fms】 http://127.0.0.1/fmsFoeLedgerHandle/queryPage
- cn.ztessc.controller.provide.FmsFoeLedgerHandleController#queryPage
- fms_foe_ledger_handle
- AUTH_TYPE_14("14"): 内部借款管理
```json
{
    "boeClassify": "fms_financing_0900011",
    "boeWaitClassify": "fms_financing_0900011",
    "query": []
}
```

```sql
-- 经办数据
select *
FROM fms_foe_ledger_handle f
WHERE f.enabled_flag = '0'
  AND f.group_id = 'ba2ad44f9aba2ad54fa43139322e0000'
  AND (
        ( f.boe_template_code IN ('TEMPLATE_INTERNAL_LOAN_REPAYMENT_APPLY_FMS','INTERNAL_LOAN_APPLICATION_FORM')
          AND f.lender_le_id IN ('0b22741bcbae86d9e6de3139322e76ad', '0b22741bcbaf4eaf93223139322e7d66','0b22741bcbae86d87b413139322e7688')
        )
        OR
        ( f.boe_template_code NOT IN ('TEMPLATE_INTERNAL_LOAN_REPAYMENT_APPLY_FMS', 'INTERNAL_LOAN_APPLICATION_FORM')
          AND ( f.lender_le_id IN ( '0b22741bcbae86d9e6de3139322e76ad', '0b22741bcbaf4eaf93223139322e7d66', '0b22741bcbae86d87b413139322e7688' ) OR f.employee_id = '0b22741bd2c9512324393139322e09ae')
        )
    )
```
