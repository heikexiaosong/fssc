### 提前支取(利息计算)
- http://127.0.0.1/fmsInnerDepositFlowRecord/getSettleInterest
```json
{
  "flowType": "3",
  "depositId": "dc581a4ec3dbff70a63c3137322e6f6c",
  "effectiveDate":  "2026-08-05",
  "amount": 300000,
  "rotationType": "BJzq"
}
```

```sql
select *
from FMS_INNER_INTERRATE_RECORD t
	inner join (
		select INNER_ID, max(effective_date) as effective_date
		from FMS_INNER_INTERRATE_RECORD t
		where 1 = 1
			and t.INNER_ID in ('468b2cbe105587e31507b1b7b2ae47e0')
			and t.effective_date <= TIMESTAMP '2026-08-05 00:00:00.000'
			and t.enabled_flag = '0'
		group by t.INNER_ID 
	) tmp
	on t.INNER_ID = tmp.INNER_ID
		and t.effective_date = tmp.effective_date
		and t.INNER_ID in ('468b2cbe105587e31507b1b7b2ae47e0')
		and t.enabled_flag = '0'
order by t.effective_date asc
```
