### 授信登记关联
- https://127.0.0.1/fmscollateralmanage/page
```json
{
  "currencyCode_EQ": "CNY",
  "leId_EQ": "782d4959f9aa3a770c3e3132372e000f",
  "facilityRegisterNo_ISNULL": 1,
  "warrantyNumber_ISNULL": 1
}
```

```sql
select *
  from fms_collateral_manage fmscollate0_
 where fmscollate0_.currency_code = 'CNY'
   and fmscollate0_.le_id = '782d4959f9aa3a770c3e3132372e000f'
   and (fmscollate0_.facility_register_no is null)
   and (fmscollate0_.warranty_number is null)
   and fmscollate0_.enabled_flag = '0'
   and fmscollate0_.group_id = '98384a38c6501ea777273132372e0008'
order by fmscollate0_.create_date desc, fmscollate0_.id desc
```
