```python

//付款方总行
String payerBankHeadCode = fmsPaymentApplyHeaderDTO.getPayerBankHeadCode();

// 1. 复制对象（基础数据）         
canImport = ConvertUtils.sourceToTarget(dto, FmsPayRollDTO.class)
    
// 2.set开户行名称(dto.getBankUnitNumber())
canImport.setBankUnitName(bankCacheDTO.getName());

// 3. 比较付款方银行行号与收款方开户行行号，设置跨行标识
// - 需求： https://www.tapd.cn/tapd_fe/67410840/story/detail/1167410840001331803
if (StrUtil.equals(bankUnitedCodeDTO.getBankTypeCode(), payerBankHeadCode)) {
    canImport.setCrossLineIdentification(FmsEnumType.CROSS_LINE_IDENTIFICATION.SAME_BNAK.getName());
} else {
    canImport.setCrossLineIdentification(FmsEnumType.CROSS_LINE_IDENTIFICATION.CROSS_BANK.getName());
}

// 4. set工资金额
canImport.setPayRoll(new BigDecimal(payRollAmount));

// 5. set工资金额
canImport.setId(UuidUtils.getUuid());

// 6. set代发工资单据id
canImport.setBoeId(boeId);

// 7. set明细单号
String orderNumber = String.format(FmsConstants.DETAILED_ORDER_NUMBER_FORMAT, ++maxDetailedOrderNumber);
canImport.setDetailedOrderNumber(orderNumber);
canImport.setBankOrderNumber(orderNumber);

// 8. set创建人
EntityUtils.setWho(canImport);
```
