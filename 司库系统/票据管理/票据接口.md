and ldgbillacc0_.enabled_flag = '0'
and ldgbillacc0_.group_id = 'af4ce58b58c721b9dcde3137322e0004'
and ldgbillacc0_.postal_order_code = 'CJRCS071009877'
        and ldgbillacc0_.ticket_system = 'new'
        and ldgbillacc0_.subticket_start = 100001
and ldgbillacc0_.subticket_start_int = 100001 
and ldgbillacc0_.subticket_end = 7100000
and ldgbillacc0_.subticket_end_int = 7100000
and ldgbillacc0_.ori_bill_code = 'CJRCS071009877'

                select count(fmsbillacc1_.id)
