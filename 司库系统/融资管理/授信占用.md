# 授信占用


### 应付票据信息写入接口
- cn.ztessc.service.bill.FmsPayBillAccountService#billOccupyCreditBreed
```java
// List<LdgBillAccountDTO> ldgBillAccountDTOList
for data in ldgBillAccountDTOList:
    useCreditQuota = data.getUseCreditQuota();
    # 不占用授信continue
    if useCreditQuota != '0':
        continue;

    creditVJson = data.getCreditVJson();
    JSONObject creditV = ZfsAliJsonUtils.parseObj(creditVJson);
    // 查询授信品种详情
    FmsBreedChildUseEntity breedChildUseEntity = ObjUtil.newInstance(FmsBreedChildUseEntity.class);
    breedChildUseEntity.setId(UuidUtils.getUuid());  
    breedChildUseEntity.setUsedDate(new Date());
   // 占用授信的业务信息 
    breedChildUseEntity.setBreedId(data.getId());             // 占用授信的业务id    
    breedChildUseEntity.setCreditAllocationChildId(data.getCreditAllocationChildId()); 
    breedChildUseEntity.setBusinessCode(data.getPostalOrderCode());    // 业务编号-汇票编码
    breedChildUseEntity.setLeId(data.getLeId());  // 核算主体 
    breedChildUseEntity.setExtendS1(data.getBillTypeName());  // 汇票类型    
    breedChildUseEntity.setExtendS2(data.getPaperTypeName());  // 汇票形式    
    breedChildUseEntity.setExtendS3(data.getPayVendorName());  // 出票人    
    breedChildUseEntity.setBenefitPerson(data.getReceiveVendorName()); // 收票人
    breedChildUseEntity.setStartDate(data.getTicketDate());
    breedChildUseEntity.setEndDate(data.getBillExpireDate());

    breedChildUseEntity.setBankId(data.getPayBank());  // 开户行    
    breedChildUseEntity.setAmount(data.getTicketAmount());    
    // 新逻辑：占用额度 = usedBankLimit（= 扣保证金前占用额度 - Σ保证金扣减，由 applyDepositDeductionForCredit 计算）
    breedChildUseEntity.setUsedLimit(data.getUsedBankLimit() ? data.getTicketAmount());
    
    // 占用类型：占用: OCCUPY("1", "占用"); RELEASE("2", "释放"); CARRY_OCCUPY("3", "结转占用"); CARRY_RELEASE("4", "结转释放")
    breedChildUseEntity.setUsedType("1");
    // 授信信息 授信品种-票据： BILL("13", "票据")
    breedChildUseEntity.setBreedType("13");
    
    breedChildUseEntity.setCreditId(creditV.getString("creditId"));
    breedChildUseEntity.setCreditBreedId(creditV.getString("breedId"));
    breedChildUseEntity.setCreditBreedChildId(creditV.getString("breedChildId"));
    breedChildUseEntity.setCreditOccupyType(creditV.getString("occupType"));
    breedChildUseEntity.setCurrencyCode(creditV.getString("currencyCode"));
    
    breedChildUseEntityList.add(breedChildUseEntity);        
```
