recodeSource: 来源系统， FMS快速编码： DATA_SOURCE
list：
  operationSeqNum： 操作流水号 required


payerInnerAccountNumList： [list::payerInnerAccountNum]

select * from FMS_INNER_ACCOUNT where BANK_NUMBER in [list::payerInnerAccountNum]
fmsInnerAccountMap： <BankNumber, FmsInnerAccountEntity>

-- 操作流水号信息
select * from FMS_ROAD_AMOUNT_RECORD where OPERATION_SEQ_NUM in [list::operationSeqNum]
fmsRoadAmountRecordMap: <OperationSeqNum, FmsRoadAmountRecordEntity>

-- 内部户在途明细信息
select * from FMS_ROAD_AMOUNT_DETAIL where OPERATION_SEQ_NUM in [list::operationSeqNum]
