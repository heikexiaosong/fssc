# 刷新付款申请单支付状态


### 【定时器】
- cn.ztessc.service.task.JobCenter#refreshPaymentApplyStatus
```sql
-- paymentInstructionsDTOs: 待处理数据
select fpi.*, fpab.BILL_ID, fpah.PAYMENT_CURRENCY as HEARD_PAYMENT_CURRENCY, fpah.PAYER_BANK_ACCOUNT_NUM as HEARD_PAYER_BANK_ACCOUNT_NUM, fpah.APPLY_AMOUNT as HEARD_APPLY_AMOUNT
  from zfs_fms.FMS_PAYMENT_APPLY_BILL fpab 
       left join zfs_fms.fms_payment_instructions fpi on fpab.BOE_ID = fpi.BOE_ID 
       left join zfs_fms.fms_payment_apply_header fpah on fpab.BOE_ID = fpah.ID
 where fpab.ENABLED_FLAG = '0' 
   and fpi.ENABLED_FLAG = '0' 
   and fpah.ENABLED_FLAG = '0'
   and (fpi.INSTRUCTION_NO is not null or fpi.INSTRUCTION_NO != '') 
   and (fpi.BOE_ID is not null or fpi.BOE_ID != '') 
   and (fpab.BILL_ID is not null or fpab.BILL_ID != '') 
   and (fpah.APPLY_AMOUNT is not null or fpah.APPLY_AMOUNT != '') 
   and fpi.PAYMENT_STATUS = '10'
   and fpi.INSTRUCTION_TYPE = '3'
```
