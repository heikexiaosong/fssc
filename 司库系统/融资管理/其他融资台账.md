# 用信额度占用记录
# FmsCreditUsageRecordDTO processCommonRecordFields(String creditId, String creditCode, String currency, String financeAmountStr, String leId)
# FmsFinanceLedgerFactoringExcelDTO dto: dto.getId(), dto.getFinanceNo(), dto.getCurrencyCode(), dto.getFinanceAmount(), dto.getLeId()
FmsCreditUsageRecordDTO creditUsageRecord = ObjUtil.newInstance(FmsCreditUsageRecordDTO.class);
creditUsageRecord.setId(UuidUtils.getUuid());
creditUsageRecord.setCreditId(creditId);
creditUsageRecord.setCreditCode(creditCode);
BigDecimal financeAmount = BigDecimal.ZERO;
if (StrUtil.isNotBlank(financeAmountStr)) {
    financeAmount = new BigDecimal(financeAmountStr);
}
creditUsageRecord.setCreditAmount(financeAmount);
creditUsageRecord.setCurrencyCode(currency);
creditUsageRecord.setCreditLeId(leId);
# FmsEnumType.CREDIT_MODE.CREDIT_MODE_05("05"): 债券和其它融资
creditUsageRecord.setCreditMode("05");
# FmsEnumType.CREDIT_USED_TYPE.OCCUPY("1", "占用")
creditUsageRecord.setUsedType("1");
creditUsageRecord.setUsedDate(new Date());

# 用信金额 设置为  usedBankLimit(本次用信占用额度)
creditUsageRecord.setCreditAmount(new BigDecimal(quotaDTO.getUsedBankLimit()));
# 增加设置用信类型ID
creditUsageRecord.setUsedCreditType(quotaDTO.getUsedCreditType());
creditUsageRecord.setUsedCreditNo(quotaDTO.getUsedCreditNo());
