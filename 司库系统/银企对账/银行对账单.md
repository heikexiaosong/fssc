#### 余额调节表关联银行对账单(balanceAdjustmentLinkBankStatement)


### 提取
- http://127.0.0.1/fmsreconciliationhandle/extract
```json
{
  "bankNumber": "648354523",
  "lastReconDate": "2026-06-01",
  "transactionDateStart": "2026-06-01",
  "transactionDateEnd": "2026-06-30",
  "currency": "CNY",
  "leId": "df668b4d55989edf3c5eb1b7b2ae1040",
  "balanceAdjustment": "Y",
  "reconciliationMode": "M"
}
```

```sql
select *
  from fms_reconciliation_flow  
 where bank_number = '648354523' 
   and currency = 'CNY' 
   and reconciliation_flow_status = '1' 
   and enabled_flag = '0'

-- 
select count(id) as col_0_0_ 
  from fms_balance_adjustment  
 where bank_number = '648354523' 
   and currency = 'CNY' 
   and status in ('1', '2', '4') 
   and enabled_flag = '0'

-- 
select *
  from fms_reconciliation_account 
 where bank_number = '648354523' 
   and currency = 'CNY' 
   and enabled_flag = '0' 
limit 2

-- 
select t.* 
  from FMS_ERP_TRANSACTION t 
 where CURRENCY_CODE = 'CNY' 
   and BANK_NUMBER = '648354523' 
   and enabled_flag = '0' 
   and ((TRANSACTION_DATE >= '2026-06-01' and RECONCILIATION_STATUS in ('1', '5')) 
        or RECONCILIATION_STATUS = '5')   -- 1: 未对账; 2: 对账中; 3: 已勾兑; 4: 已对账; 5: 未达账
   and TRANSACTION_DATE <= '2026-06-30'

--
select t.* 
  from FMS_TRANSACTION_DETAILS_HIS t 
 where TRANSACTION_CURRENCY_CODE = 'CNY' 
   and BANK_NUMBER = '648354523' 
   and enabled_flag = '0' 
   and ((TRANSACTION_DATE >= '2026-06-01' and RECONCILIATION_STATUS in ('1', '5')) 
   or RECONCILIATION_STATUS = '5') 
   and TRANSACTION_DATE <= '2026-06-30 23:59:59.000'
```



### 银行账查询-银企对账经办页面
- http://127.0.0.1/fmsreconciliationhandle/bankData
```json
{
  "blendType": "1",
  "blendBatchNo": "20260719194022851"
}
```

```sql
select *
  from fms_erp_transaction 
 where blend_batch_no = '20260719194022851' 
	 and blend_type in ('1') 
	 and enabled_flag = '0' 
order by blend_type asc, transaction_date asc, id asc 
limit 20
```


### 企业账查询-银企对账经办页面
- http://127.0.0.1/fmsreconciliationhandle/erpData
```json
{
  "blendType": "1",
  "blendBatchNo": "20260719194022851"
}
```

```sql
select *
  from fms_transaction_details_his  
 where blend_batch_no = '20260719194022851' 
	 and blend_type in ('1')  -- 1: 未勾兑; 2: 已勾兑; 3: 手动勾兑
order by blend_type asc, transaction_date asc, id asc 
limit 20
```
