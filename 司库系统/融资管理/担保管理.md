### 担保占用节点
- 【【黑交投司库1226】贷款登记节点占用授信，放款节点占用担保】https://www.tapd.cn/tapd_fe/67410840/story/detail/1167410840001318975
- 系统参数  COLLATERAL_OCCUPANCY_NODE：备注：“担保占用节点，0-登记，1-放款”，默认 0
- 进行贷款台账录入保存时，对参数值进行记录
  1. 如果参数值≠1，则该贷款按照0的逻辑执行，
  2. 如果参数值=1，则该贷款按照1的逻辑执行；
  3. 贷款台账执行中途修改了参数值，不影响已经生成的贷款数据的逻辑。
- 担保占用节点，0-登记，1-放款，参数为0，在贷款台账新增担保占用节点字段，记录为0，在贷款台账录入占用担保(注意区分银团和非银团)
- 担保占用节点，0-登记，1-放款，参数为1，在贷款台账新增担保占用节点字段，记录为1，在贷款台账放款占用担保，还款释放担保(注意区分银团和非银团)


### 担保占用逻辑（融资贷款登记）
- cn.ztessc.service.provide.FmsProvideHeaderService#dealFacilityRegisterNo
- collateralOccupancyNode != 1

```python
# 担保占用记录 FMS_GUARANTEE_USE
FmsGuaranteeUseEntity guaranteeUseEntity = ObjUtil.newInstance(FmsGuaranteeUseEntity.class);
guaranteeUseEntity.setId(UuidUtils.getUuid());
guaranteeUseEntity.setBusId(busId);         # 业务ID： 贷款台账id
guaranteeUseEntity.setBusCode(busCode);     # 业务编码： 贷款台账编码
guaranteeUseEntity.setBusType("0");         # 融资类型 FmsEnumType.GUARANTEE_BUS_TYPE.TYPE_0("0", "贷款")
guaranteeUseEntity.setGuaranteeId(guaranteeId); # 担保id
guaranteeUseEntity.setUsedLimit(usedLimit);     # 占用额度
guaranteeUseEntity.setUsedDate(new Date());     # 占用时间
# 占用类型： FmsEnumType.CREDIT_USED_TYPE.OCCUPY("1", "占用"); RELEASE("2", "释放"); CARRY_OCCUPY("3", "结转占用"); CARRY_RELEASE("4", "结转释放")
guaranteeUseEntity.setUsedType("1");

# 对内担保 FMS_INTERNAL_GUARANTEE
BigDecimal availableQuota = entity.getAvailableQuota();
BigDecimal occupyQuota = entity.getOccupyQuota();
# 可用-额度，占用+额度
availableQuota = availableQuota.subtract(guaranteeLimit);
occupyQuota = occupyQuota.add(guaranteeLimit);
entity.setAvailableQuota(availableQuota);
entity.setOccupyQuota(occupyQuota);
# 担保关联标识
entity.setFacilityRegisterNo(busId);
```

# 担保占用记录 FMS_GUARANTEE_USE

guaranteeUseEntity.setUsedDate(new Date());     # 占用时间
# 占用类型： FmsEnumType.CREDIT_USED_TYPE.OCCUPY("1", "占用"); RELEASE("2", "释放"); CARRY_OCCUPY("3", "结转占用"); CARRY_RELEASE("4", "结转释放")
guaranteeUseEntity.setUsedType("2");

# 对内担保 FMS_INTERNAL_GUARANTEE
update FMS_INTERNAL_GUARANTEE 
   set AVAILABLE_QUOTA = AVAILABLE_QUOTA + :userdLimit, OCCUPY_QUOTA = OCCUPY_QUOTA - :userdLimit
 where id = :guaranteeId
